	type = character

	warning_level = 0.70

	# Plotter scope
	potential = {
		has_character_flag = flag_liberation_revolt
		is_landed = yes
		higher_real_tier_than = BARON
		is_adult = yes
		independent = no
		prisoner = no
		is_actually_incapable_trigger = no
		holy_order = no
	}

	# Target scope
	allow = {
		always = yes
	}

	# Faction member scope
	# ROOT is joiner
	# FROM is target
	allow_join = {
		ROOT = {
			is_ruler = yes
		#	is_adult = yes
		#	independent = no
		#	is_landed = yes
		#	is_incapable = no
		#	prisoner = no
		}
	}

	# AI creation weight
	# FROM is faction creator
	# ROOT is title scope
	chance = {
		factor = 1000

		trigger	= {
			FROM = { has_character_flag = flag_liberation_revolt }
		}
	}

	# AI membership weight
	# ROOT is the prospective member
	# FROM is the faction leader
	# FROMFROM is the target title or character
	membership = {
		factor = 1000

		trigger = {
			culture = FROM
		}
	}

	success = {
		independent = yes
	}

	effect = {
		FROM = {
			any_faction_backer = {
				faction = faction_liberation_revolt

				reverse_opinion = {

					who = FROM
					years = 5
				}
			}
		}
	}

	abort = {
		always = no	# Factions will abort if the Potential or Allow trigger is no longer valid
	}
