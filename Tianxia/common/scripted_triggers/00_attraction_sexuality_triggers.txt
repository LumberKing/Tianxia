# has_any_sex_interest_trait_trigger

has_any_sexuality_trait_trigger = {
	OR = {
		trait = heterosexual
		trait = homosexual
		trait = bisexual
		trait = asexual
	}
}

has_any_sex_interest_trait_trigger = {
	OR = {
		trait = sex_interest_female
		trait = sex_interest_male
	}
}

has_any_romance_interest_trait_trigger = {
	OR = {
		trait = romance_interest_female
		trait = romance_interest_male
	}
}
