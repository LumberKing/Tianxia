is_valid_claimant_prev_no_leader_trigger = {
	PREV = {
		# Don't faction for duchies/counties if the holder is of a sufficiently high tier
		OR = {
			holder_scope = {
				real_tier = PREV
			}
			tier = king
		}
	}
	
	trigger_if = {
		limit = { is_female = no }

		NOT = {
			PREV = {
				succ_law_title = {
					has_law = enatic_succession
				}
			}
		}
	}
	
	trigger_else = {
		NOT = {
		
			PREV = {
				succ_law_title = {
					has_law = agnatic_succession
				}
			}
		}
	}
	
	NAND = {
		controls_religion = yes
		is_theocracy = yes
	}

	# The Pope, static mercs and holy orders shouldn't get titles at or above their tier
	trigger_if = {
		limit = {
			is_ruler = yes

			primary_title = {
				is_primary_type_title = yes
			}
		}

		lower_real_tier_than = PREV
	}
}

is_valid_claimant_prev_trigger = {
	event_target:potential_faction_leader = {
		trigger_if = {
			limit = {
				liege = {
					is_nomadic = yes
				}
			}

			character = new_character # Claimant
			has_dlc = "Horse Lords"
			is_nomadic = yes
			NOT = { has_blood_oath_with = liege }
		}
	}
	
	PREV = {
		trigger_if = {
			limit = {
				holder_scope = {
					OR = {
						is_the_tenno_trigger = yes
						is_ryukyuan_tenno_trigger = yes
					}
				}
				event_target:potential_faction_leader = {
					could_have_japanese_government_trigger = yes
				}
			}
			
			PREV = {
				OR = {
					is_the_tenno_trigger = yes
					is_ryukyuan_tenno_trigger = yes
				}
				
				character = event_target:potential_faction_leader
			}
		}
		
		# Don't faction for duchies/counties if the holder is of a sufficiently high tier
		OR = {
			holder_scope = {
				real_tier = ROOT
			}
			tier = king
		}
		
		# Don't faction for titles you don't care about!
		OR = {
			AND = {
				holder_scope = {
					NOT = {
						any_demesne_title = {
							is_primary_holder_title_tier = yes # Changed from is_primary_holder_title, since secondary titles might matter
							de_jure_vassal_or_below = event_target:potential_faction_leader
						}
					}
				}
				is_primary_holder_title = yes
			}
			de_jure_vassal_or_below = event_target:potential_faction_leader
			PREV = {
				character = event_target:potential_faction_leader
			}
		}
		
		OR = {
			PREV = {
				character = event_target:potential_faction_leader
			}
			holder_scope = {
				OR = {
					NOT = {
						government = chinese_imperial_government
					}
					dynasty = PREVPREV
					has_character_modifier = mandate_of_heaven_2
					has_character_modifier = mandate_of_heaven_1
				}
			}
		}
	}
	
	OR = {
		same_realm = event_target:potential_faction_leader

		event_target:potential_faction_leader = {
			liege = { independent = yes }
		}
	}

	trigger_if = {
		limit = { is_female = no }

		NOR = {
			event_target:potential_faction_leader = {
				OR = {
					has_religion_feature = religion_matriarchal
					has_religion_feature = religion_feature_minangkabau
				}
			}

			PREV = {
				succ_law_title = {
					has_law = enatic_succession
				}
			}
		}
	}
	
	trigger_else = {
		NOR = {
			event_target:potential_faction_leader = {
				OR = {
					has_religion_feature = religion_patriarchal
					has_religion_feature = religion_feature_batak
					has_religion_feature = religion_feature_yi
					AND = {
						religion = confucian
						gender_equality_trigger = no
						religions_are_standard_trigger = yes
					}
				}
			}
			
			PREV = {
				succ_law_title = {
					has_law = agnatic_succession
				}
			}
		}
	}

	NOR = {
		has_landed_title = PREV
		is_liege_or_above_of = event_target:potential_faction_leader
		AND = {
			controls_religion = yes
			is_theocracy = yes
		}
	}

	# The Pope, static mercs and holy orders shouldn't get titles at or above their tier
	trigger_if = {
		limit = {
			is_ruler = yes

			primary_title = {
				is_primary_type_title = yes
			}
		}

		lower_real_tier_than = PREV
	}

	# Claimant isn't already fighting for claim on this title
	event_target:potential_faction_leader = {
		liege = {
			NOT = {
				defending_against_claimant = {
					character = PREVPREV
					title = PREVPREVPREV
				}
			}
		}
	}
	
	# The Tenno is safe from regular claims...
	trigger_if = {
		limit = {
			event_target:potential_faction_leader = {
				liege = {
					is_the_tenno_trigger = yes
				}
			}
		}
		
		OR = {
			PREV = { # ... unless it's just a kingdom...
				tier = king
				holder_scope = {
					real_tier = emperor
					NOT = {
						dynasty = PREVPREV
					}
				}
			}
			event_target:potential_faction_leader = { # ... or we don't care...
				could_have_japanese_government_trigger = no
			}
			event_target:potential_faction_leader = { # ... or we're the knockoff
				character = PREV
				is_ryukyuan_tenno_trigger = yes
			}
		}
	}
	
	# The Ryukyuan knockoff is safe from regular claims...
	trigger_else_if = {
		limit = {
			event_target:potential_faction_leader = {
				liege = {
					is_ryukyuan_tenno_trigger = yes
				}
			}
		}
		
		OR = {
			PREV = { # ... unless it's a lower title
				holder_scope = {
					real_tier = emperor
				}
				tier = king
			}
			any_owned_bloodline = { # or it's a bloodline member...
				has_bloodline_flag = divine_ancestry_bloodline_ryukyuan
			}
			event_target:potential_faction_leader = { # ... or we don't care...
				could_have_japanese_government_trigger = no
			}
			event_target:potential_faction_leader = { # ... or we're the Tenno
				character = PREV
				is_the_tenno_trigger = yes
			}
		}
	}
	
	# The Shogun shouldn't face claimant factions for the Tenno's claim
	trigger_else_if = {
		limit = {
			event_target:potential_faction_leader = {
				liege = {
					has_japanese_government_trigger = yes
				}
				tier = emperor
			}
		}
		
		is_the_tenno_trigger = no
		
		NOT = {
			any_dynasty_member = {
				is_the_tenno_trigger = yes
			}
		}
	}
	
	# Avoid putting women on the throne of China/etc. under normal conditions
	trigger_else_if = {
		limit = {
			is_female = yes
			PREV = {
				holder_scope = {
					government = chinese_imperial_government
					is_female = no
				}
				succ_law_title = {
					has_law = agnatic_succession
				}
				tier = emperor
			}
		}
		
		event_target:potential_faction_leader = {
			OR = {
				is_female = yes
				AND = {
					could_have_chinese_government_trigger = no
					OR = {
						could_have_japanese_government_trigger = no
						has_religion_feature = religion_equal
						has_religion_feature = religion_matriarchal
					}
				}
			}
		}
	}
}
