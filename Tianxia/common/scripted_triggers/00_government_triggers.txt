# Merchant Republic
merchant_republic_government_potential_trigger = {
	republic_group_potential_trigger = yes
	trigger_if = { # Established MR requiremends
		limit = {
			is_merchant_republic = yes
			has_game_started = yes
		}

		# Require at least one province and be Duke tier or higher
		higher_real_tier_than = COUNT
		num_of_count_titles >= 1

		# MR can't be under MR
		trigger_if = {
			limit = { independent = no }

			liege_before_war = {
				is_merchant_republic = no
			}
		}
	}
	trigger_else_if = { # Established Patrician requirements
		limit = {
			is_patrician = yes
			has_game_started = yes
		}

		# Liege must be MR
		liege_before_war = {
			is_merchant_republic = yes

			# MR can't be under MR
			trigger_if = {
				limit = { independent = no }

				liege_before_war = {
					is_merchant_republic = no
				}
			}
		}
	}
	trigger_else = { # Pre-game and new MR requirements
		OR = {
			is_patrician = yes

			trigger_if = {
				limit = { capital_scope = { is_coastal = yes } }

				# Require at least one province and be Duke tier or higher
				higher_real_tier_than = COUNT
				num_of_count_titles >= 1

				# MR can't be under MR
				trigger_if = {
					limit = { independent = no }

					liege_before_war = {
						is_merchant_republic = no
					}
				}
			}
		}
	}
}

merchant_republic_government_allow_trigger = {
	republic_group_allow_trigger = yes
}

# Theocratic Group
# In case someone wants to add more Theocracies for some reason
theocratic_group_potential_trigger = {
	holy_order = no
	is_patrician = no

	trigger_if = {
		limit = { controls_religion = yes }

		could_be_temporal_religious_head_trigger = no
	}
	trigger_else = {
		NOR = {
			gets_muslim_governments_trigger = yes
			is_government_potential = japanese_monastic_feudal_government
			is_government_potential = monastic_feudal_government
			is_government_potential = holy_mandate_government
		}
	}

	trigger_if = {
		limit = { is_female = yes }
		religion_allows_female_temple_holders = yes
	}
	trigger_else = {
		religion_allows_male_temple_holders = yes
	}
	NOT = {
		trait = nomadic_trait
	}
}

theocratic_group_allow_trigger = {}

theocratic_government_potential_trigger = {
	theocratic_group_potential_trigger = yes
}

	theocratic_group_allow_trigger = yes
}

# Tribal Group
# In case someone wants to add more Tribals for some reason
tribal_group_potential_trigger = {
	mercenary = no
	is_patrician = no
	holy_order = no
	NOT = { is_government_potential = order_government }

	trigger_if = {
		limit = { has_game_started = yes }
		is_feudal = no
	}

	could_be_temporal_religious_head_trigger = yes # If holder is a religion head, makes sure they shouldn't be a theocracy instead
}

tribal_group_allow_trigger = {}

has_steppe_nomad_culture_trigger = {
	OR = {
		AND = {
			culture_group = altaic
			NOT = {
				culture = jurchen
			}
		}
		culture = alan
		culture = hungarian
	}
}

	OR = {
		culture = bedouin
		culture = berber
		culture = levantine
		culture = somali # TIANXIATODO: East African?
	}
}

# Nomadic Tribal Successor State
nomadic_tribal_successor_state_government_potential_trigger = {
	tribal_group_potential_trigger = yes
	has_dlc = "Horse Lords"
	has_steppe_nomad_culture_trigger = yes
	higher_real_tier_than = duke
	trait = nomadic_trait
	OR = {
		AND = {
			is_save_game = no
			has_game_started = no
			primary_title = {
				OR = {
					title = e_golden_horde
					title = e_chagatai
					AND = {
						title = k_mongolia
						year > 1220
					}
				}
				# TIANXIATODO: Other applicable
			}
		}
		any_owned_bloodline = {
			has_bloodline_flag = genghis_khan
		}
	}
	NOT = {
		is_government_potential = great_khan_government
	}
}

nomadic_tribal_successor_state_government_allow_trigger = {
	tribal_group_allow_trigger = yes
}
