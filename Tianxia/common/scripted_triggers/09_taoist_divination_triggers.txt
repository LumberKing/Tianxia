has_a_diviner_modifier_trigger = {
	OR = {
		has_character_modifier = fulfilled_divination_great
		has_character_modifier = fulfilled_divination_ok
		has_character_modifier = fulfilled_divination_bad
		has_character_modifier = fulfilled_divination_worst
	}
}

has_a_taoist_divination_modifier_trigger = {
	OR = {
		has_character_modifier = good_battle_taoist_divination
		has_character_modifier = good_relationship_taoist_divination
		has_character_modifier = good_power_taoist_divination
		has_character_modifier = bad_battle_taoist_divination
		has_character_modifier = bad_relationship_taoist_divination
		has_character_modifier = bad_power_taoist_divination
	}
}
