valid_romantic_match_ROOT_trigger = {
	custom_tooltip = {
		text = valid_romantic_match_ROOT
		trigger_if = {
			limit = {
				is_female = yes
			}
			
			ROOT = {
				trait = romance_attraction_female
				trait = romance_interest_female
			}
		}
		
		trigger_else = {
			ROOT = {
				trait = romance_attraction_male
				trait = romance_interest_male
			}
		}
		
		trigger_if = {
			limit = {
				ROOT = {
					is_female = yes
				}
			}
			
			trait = romance_attraction_female
			trait = romance_interest_female
		}
		
		trigger_else = {
			trait = romance_attraction_male
			trait = romance_interest_male
		}
		
		no_incest_concern_ROOT_trigger = yes
	}
}

		text = valid_sexual_interest_FROM
		trigger_if = {
			limit = {
				is_female = yes
			}
			
			FROM = {
				trait = sex_interest_female
			}
		}
		
		trigger_else = {
			FROM = {
				trait = sex_interest_male
			}
		}
		
		trigger_if = {
			limit = {
				FROM = {
					is_female = yes
				}
			}
			
			trait = sex_interest_female
		}
		
		trigger_else = {
			trait = sex_interest_male
		}
		
		no_incest_concern_FROM_trigger = yes
	}
}

no_incest_concern_ROOT_trigger = {
	trigger_if = {
		limit = {
			is_close_relative = ROOT
		}
		
		trigger_if = {
			limit = {
				has_secret_religion = yes
				true_religion = ROOT
			}
			
			OR = {
				AND = {
					true_religion_has_incest = yes
					ROOT = {
						true_religion_has_incest = yes
					}
				}
				AND = {
					religions_are_standard_trigger = yes # TIANXIATODO: Better handling
					sibling = ROOT
					true_inca_religion_trigger=  yes
					ROOT = {
						true_inca_religion_trigger = yes
					}
				}
			}
		}
		
		trigger_else = {
			OR = {
				AND = {
					religion_has_incest = yes
					ROOT = {
						religion_has_incest = yes
					}
				}
				AND = { # Breaking religious norms is not a problem...
					is_devil_worshiper_trigger = yes
					same_society_as = ROOT
				}
				AND = {
					religions_are_standard_trigger = yes # TIANXIATODO: Better handling
					sibling = ROOT
					religion_openly_inca_or_reformed_trigger = yes
					ROOT = {
						religion_openly_inca_or_reformed_trigger = yes
					}
				}
			}
		}
	}
}
