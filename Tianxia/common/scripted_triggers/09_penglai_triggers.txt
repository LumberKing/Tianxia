# is_eligible_for_penglai_trigger
# can_reach_penglai_trigger
# can_return_to_sender_from_penglai_trigger

is_eligible_for_penglai_trigger = {
	gets_chinese_governments_trigger = yes
	can_reach_penglai_trigger = yes
	higher_real_tier_than = duke
	in_revolt = no
	is_inaccessible_or_incapable_trigger = no
	is_actual_immortal_trigger = no
}

can_reach_penglai_trigger = {
	custom_tooltip = {
		text = can_reach_penglai
		
		any_realm_province = {
			OR = {
				county = {
					title = c_hangzhou
				}
				can_naval_path_to = c_hangzhou
			}
		}
	}
}

can_return_to_sender_from_penglai_trigger = {
	event_target:penglai_sender = {
		is_landed = yes
		prisoner = no
		is_actually_incapable_trigger = no
		OR = {
			character = ROOT
			any_realm_lord = {
				character = ROOT
			}
		}
	}
}
