		trait = zodiac_ox

		trait = zodiac_rabbit
		trait = zodiac_dragon
		trait = zodiac_snake
		trait = zodiac_horse
		trait = zodiac_goat
		trait = zodiac_monkey
		trait = zodiac_rooster
		trait = zodiac_dog
		trait = zodiac_pig
	}
}

		trait = element_wood
