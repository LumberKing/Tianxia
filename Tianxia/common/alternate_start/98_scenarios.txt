nomads = { # WIP
	type = checkbox
	sort_priority = 999

	checked = {
		effect = {
			any_province = {
				limit = {
					any_province_holding = {
						holding_type = tribal
					}
					holder_scope = {
						OR = {
							has_steppe_nomad_culture_trigger = yes
							has_desert_nomad_culture_trigger = yes
						}
					}
				}
				
				holder_scope = {
					log = "I should be nomadic tribal!"
					
					add_trait = nomadic_trait
					if = {
						limit = {
							has_steppe_nomad_culture_trigger = yes
						}
						set_government_type = steppe_nomadic_tribal_government
					}
					else = {
						set_government_type = desert_nomadic_tribal_government
					}
				}
			}
		}
	}

	unchecked = {}
}
