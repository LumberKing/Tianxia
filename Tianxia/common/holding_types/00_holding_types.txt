				}
			}

		}

			limit = {
				trade_route = trans_saharan_trade_route

					is_coastal = no
				}
			}
			trans_saharan_trade_post_trigger = yes
		}

			limit = {
				NOT = {
					has_game_rule = {

					}
				}
			}

			}
		}
	}
