set_correct_feudal_government_type_effect = {
	if = {
		limit = {
			is_government_potential = successor_state_government
		}
		set_government_type = successor_state_government
	}
	else_if = {
		limit = {
			gets_japanese_governments_trigger = yes
		}
		if = {
			limit = { is_government_potential = japanese_imperial_government }
			set_government_type = japanese_imperial_government
		}
		else_if = {
			limit = { is_government_potential = divine_imperial_government }
			set_government_type = divine_imperial_government
		}
		else_if = {
			limit = { is_government_potential = japanese_monastic_feudal_government }
			set_government_type = japanese_monastic_feudal_government
		}
		else_if = {
			limit = { is_government_potential = japanese_feudal_government }
			set_government_type = japanese_feudal_government
		}
		else = {
			set_government_type = japanese_bureaucracy_government
		}
	}
	else_if = {
		limit = {
			gets_chinese_governments_trigger = yes
		}
		if = {
			limit = {
				trait = warlord_trait
			}
			if = {
				limit = {
					is_government_potential = warlord_chinese_imperial_government
				}
				set_government_type = warlord_chinese_imperial_government
			}
			else_if = {
				limit = {
					is_government_potential = warlord_imperial_government
				}
				set_government_type = warlord_imperial_government
			}
			else = {
				set_government_type = warlord_government
			}
		}
		else = {
			if = {
				limit = {
					is_government_potential = chinese_imperial_government
				}
				set_government_type = chinese_imperial_government
			}
			else_if = {
				limit = {
					is_government_potential = eastern_imperial_government
				}
				set_government_type = eastern_imperial_government
			}
			else = {
				set_government_type = eastern_bureaucracy_government
			}
		}
	}
	else_if = {
		limit = {
			gets_muslim_governments_trigger = yes
		}
		if = {
			limit = {
				is_government_potential = caliphal_government
			}
			set_government_type = caliphal_government
		}
		else_if = {
			limit = {
				is_government_potential = ghazi_government
			}
			set_government_type = ghazi_government
		}
		else = {
			set_government_type = muslim_bureaucracy_government
		}
	}
	else_if = {
		limit = {
			is_government_potential = mandala_government
		}
		set_government_type = mandala_government
	}
	else_if = {
		limit = {
			is_government_potential = dharmic_feudal_government
		}
		set_government_type = dharmic_feudal_government
	}
	else_if = {
		limit = {
			is_government_potential = monastic_feudal_government
		}
		set_government_type = monastic_feudal_government
	}
	else_if = {
		limit = {
			is_government_potential = roman_imperial_government
		}
		set_government_type = roman_imperial_government
	}
	else_if = {
		limit = {
			is_government_potential = imperial_bureaucracy_government
		}
		set_government_type = imperial_bureaucracy_government
	}
	else_if = {
		limit = {
			is_government_potential = bureaucracy_government
		}
		set_government_type = bureaucracy_government
	}
	else_if = {
		limit = {
			is_government_potential = holy_mandate_government
		}
		set_government_type = holy_mandate_government
	}
	else = {
		set_government_type = feudal_government
	}
}

set_correct_theocratic_government_type_effect = {
	if = {
		limit = {
			gets_muslim_governments_trigger = yes
		}
		if = {
			limit = {
				is_government_potential = caliphal_government # Should be impossible
			}
			set_government_type = caliphal_government
		}
		else_if = {
			limit = {
				is_government_potential = ghazi_government # Should be impossible
			}
			set_government_type = ghazi_government
		}
		else = {
			set_government_type = muslim_bureaucracy_government
		}
	}
	else_if = {
		limit = { is_government_potential = holy_mandate_government }
		set_government_type = holy_mandate_government
	}
	else_if = {
		limit = { is_government_potential = mandala_government }
		set_government_type = mandala_government
	}
	else_if = {
		limit = { is_government_potential = monastic_feudal_government }
		set_government_type = monastic_feudal_government
	}
	else_if = {
		limit = { is_government_potential = japanese_monastic_feudal_government } # Most likely impossible
		set_government_type = japanese_monastic_feudal_government
	}
	else = {
		set_government_type = theocracy_government
	}
}
