	# First remove any rivalry
	if = {
		limit = {
			is_rival = event_target:changing_relationship_target
		}
		remove_rival = event_target:changing_relationship_target
	}
	# Then remove any rivalry modifiers
	ex_rival_opinions_cleanup_effect = yes
	# Then remove any modifiers that should not be present on the lovers
