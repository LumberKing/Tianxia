					save_event_target_as = player_heir_option_1
				}
				
				else_if = {
					limit = {
						event_target:player_heir_option_2 = {
							always = no
						}
					}
					
					save_event_target_as = player_heir_option_2
				}
				
				else = {
					save_event_target_as = player_heir_option_3
				}
				
				save_event_target_as = player_heir_option_society
			}
			
			set_character_flag = player_heir_option_society
		}
		1 = {
			# No target: Fallback
		}
	}
