spawn_peasant_army_effect = {
	# Spawn some commanders, if necessary
	while = {
		limit = { can_grant_title = title_commander }
		
		if = {
			limit = { gender_equality_no_laws_trigger = yes }
			create_character = {
				random_traits = yes
				dynasty = none
				religion = event_target:target_province
				culture = event_target:target_province
				female = 50
				age = 23
				add_trait = peasant_leader
				add_trait = tough_soldier
				flag = rebel_commander
			}
		}
		else_if = {
			limit = {
				OR = {
					has_religion_feature = religion_matriarchal
					has_religion_feature = religion_female_preference
				}
			}
			create_character = {
				random_traits = yes
				dynasty = none
				religion = event_target:target_province
				culture = event_target:target_province
				female = yes
				age = 23
				add_trait = peasant_leader
				add_trait = tough_soldier
				flag = rebel_commander
			}
		}
		else = {
			create_character = {
				random_traits = yes
				dynasty = none
				religion = event_target:target_province
				culture = event_target:target_province
				female = no
				age = 23
				add_trait = peasant_leader
				add_trait = tough_soldier
				flag = rebel_commander
			}
		}

		new_character = {
			random_list = {
				1 = { }
				1 = { add_age = 1 }
				1 = { add_age = 2 }
				1 = { add_age = 3 }
				1 = { add_age = 4 }
				1 = { add_age = 5 }
				1 = { add_age = 6 }
				1 = { add_age = 7 }
				1 = { add_age = 8 }
			}

			give_minor_title = title_commander
		}
	}

	if = {
		limit = {
			has_game_rule = {
				name = provincial_revolt_strength
				value = default
			}
		}

		random_list = {
			1 = {
				spawn_peasant_army_scale_075_effect = yes

				# Additional troops disband on peace
				while = {
					count = 2

					spawn_peasant_army_scale_075_disbanding_effect = yes
				}
			}

			1 = {
				spawn_peasant_army_scale_1_effect = yes

				# Additional troops disband on peace
				while = {
					count = 2

					spawn_peasant_army_scale_05_disbanding_effect = yes
				}
			}
		}
	}
	else = {
		# Same as above, but no attrition
		random_list = {
			1 = {
				spawn_peasant_army_scale_075_no_attrition_effect = yes

				# Additional troops disband on peace
				while = {
					count = 2

					spawn_peasant_army_scale_075_no_attrition_disbanding_effect = yes
				}
			}

			1 = {
				spawn_peasant_army_scale_1_no_attrition_effect = yes

				# Additional troops disband on peace
				while = {
					count = 2

					spawn_peasant_army_scale_05_no_attrition_disbanding_effect = yes
				}
			}
		}
	}
}
