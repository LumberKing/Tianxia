increase_regent_powers_effect = {
	primary_title = {
		trigger_switch = {
			on_trigger = has_law
			
			regency_power_law_0 = { add_law = regency_power_law_1 }
			regency_power_law_1 = { add_law = regency_power_law_2 }
			regency_power_law_2 = { add_law = regency_power_law_3 }
			regency_power_law_3 = { add_law = regency_power_law_4 }
			regency_power_law_4 = { add_law = regency_power_law_5 }
			regency_power_law_5 = { add_law = regency_power_law_6 }
		}
	}
}

increase_regent_powers_twice_effect = {
	increase_regent_powers_effect = yes
	increase_regent_powers_effect = yes
}

decrease_regent_powers_effect = {
	primary_title = {
		trigger_switch = {
			on_trigger = has_law
			
			regency_power_law_6 = { add_law = regency_power_law_5 }
			regency_power_law_5 = { add_law = regency_power_law_4 }
			regency_power_law_4 = { add_law = regency_power_law_3 }
			regency_power_law_3 = { add_law = regency_power_law_2 }
			regency_power_law_2 = { add_law = regency_power_law_1 }
			regency_power_law_1 = { add_law = regency_power_law_0 }
		}
	}
}

update_laws_for_regency_power_effect = { # TIANXIATODO: Update if council laws are added/removed
	# Note: This is only supposed to erode the liege's power, so no depowering the council if the regent is weakened!
	trigger_switch = {
		on_trigger = has_law
		
		regency_power_law_0 = {
			update_voting_laws_for_regency_power_0_effect = yes
		}
		regency_power_law_1 = {
			update_voting_laws_for_regency_power_0_effect = yes
		}
		regency_power_law_2 = {
			update_voting_laws_for_regency_power_0_effect = yes
		}
		regency_power_law_3 = {
			update_voting_laws_for_regency_power_0_effect = yes
		}
		regency_power_law_4 = {
			update_voting_laws_for_regency_power_0_effect = yes
		}
		regency_power_law_5 = {
			update_voting_laws_for_regency_power_0_effect = yes
		}
		regency_power_law_6 = {
			update_voting_laws_for_regency_power_0_effect = yes
		}
	}
	
	if = {
		limit = {
			has_dlc = "Conclave"
		}
		
		add_law = {
			law = vassal_wars_law_0
			cooldown = no
			opinion_effect = no
		}
	}
	
	if = {
		limit = {
			has_dlc = "Charlemagne"
		}
		
		add_law = {
			law = vice_royalty_0
			cooldown = no
			opinion_effect = no
		}
	}
	
	if = {
		limit = {
			has_law = revoke_duchy_law_1
		}
		
		add_law = revoke_duchy_law_0
	}
	
	if = {
		limit = {
			has_law = revoke_title_law_1
		}
		
		add_law = revoke_title_law_0
	}
}
