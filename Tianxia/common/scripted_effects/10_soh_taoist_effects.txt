remove_taoist_divination_modifiers_effect = { # For cleaning up end-states of the Hermetics Scrying Power
	hidden_effect = { remove_character_modifier = scrying_cooldown }

	trigger_switch = {
		on_trigger = has_character_modifier

		good_battle_taoist_divination       = { remove_character_modifier = good_battle_taoist_divination }
		bad_battle_taoist_divination        = { remove_character_modifier = bad_battle_taoist_divination }
		good_relationship_taoist_divination = { remove_character_modifier = good_relationship_taoist_divination }
		bad_relationship_taoist_divination  = { remove_character_modifier = bad_relationship_taoist_divination }
		good_power_taoist_divination        = { remove_character_modifier = good_power_taoist_divination }
		bad_power_taoist_divination         = { remove_character_modifier = bad_power_taoist_divination }
	}

	clr_character_flag = positive_reaction_to_divination_flag
	clr_character_flag = negative_reaction_to_divination_flag
	clr_character_flag = positive_event_flag
	clr_character_flag = negative_event_flag

	clear_delayed_event = { id = soh.13324 }
	clear_delayed_event = { id = soh.13325 }
	clear_delayed_event = { id = soh.13326 }
}

# The positive_event_flag/negative_event_flag needs to be set somewhere in the target event in order for this to work...
# CleanSlate: Nothing in vanilla sets the positive_event_flag...
calculate_diviner_outcome_effect = {
	custom_tooltip = { text = tooltip_info_taoist_divination_tt }

	if = {
		limit = {
			trigger_if = {
				limit = { has_character_flag = positive_event_flag }
				has_positive_taoist_divination_modifier_trigger = yes
			}
			trigger_else = {
				has_character_flag = negative_event_flag
				has_negative_taoist_divination_modifier_trigger = yes
			}
		}

		upgrade_diviner_modifier_effect = yes
	}
	else = {
		downgrade_diviner_modifier_effect = yes
	}

	clr_character_flag = positive_event_flag
	clr_character_flag = negative_event_flag
}

# Modifiers gotten at end-states of the Taoist divination
upgrade_diviner_modifier_effect = {
	if = {
		limit = { has_a_diviner_modifier_trigger = yes }

		trigger_switch = {
			on_trigger = has_character_modifier

			fulfilled_vision_worst = {
				remove_character_modifier = fulfilled_vision_worst

				add_character_modifier = {
					name = fulfilled_vision_bad
					years = 7
				}
			}

			fulfilled_vision_bad = {
				remove_character_modifier = fulfilled_vision_bad

				add_character_modifier = {
					name = fulfilled_vision_ok
					years = 7
				}
			}

			fulfilled_vision_ok = {
				remove_character_modifier = fulfilled_vision_ok

				add_character_modifier = {
					name = fulfilled_vision_great
					years = 7
				}
			}

			fulfilled_vision_great = {
				add_character_modifier = {
					name = fulfilled_vision_great
					years = 7
				}
			}
		}
	}
	else = {
		add_character_modifier = {
			name = fulfilled_vision_ok
			years = 7
		}
	}
}

# Modifiers gotten at end-states of the Taoist divination
downgrade_diviner_modifier_effect = {
	if = {
		limit = { has_a_diviner_modifier_trigger = yes	}

		trigger_switch = {
			on_trigger = has_character_modifier

			fulfilled_vision_great = {
				remove_character_modifier = fulfilled_vision_great

				add_character_modifier = {
					name = fulfilled_vision_ok
					years = 7
				}
			}

			fulfilled_vision_ok = {
				remove_character_modifier = fulfilled_vision_ok

				add_character_modifier = {
					name = fulfilled_vision_bad
					years = 7
				}
			}

			fulfilled_vision_bad = {
				remove_character_modifier = fulfilled_vision_bad

				add_character_modifier = {
					name = fulfilled_vision_worst
					years = 7
				}
			}

			fulfilled_vision_worst = {
				add_character_modifier = {
					name = fulfilled_vision_worst
					years = 7
				}
			}
		}
	}
	else = {
		add_character_modifier = {
			name = fulfilled_vision_bad
			years = 7
		}
	}
}
