set_up_sexuality_romance_effect = {
	if = {
		limit = {
			NOT = {
				has_character_flag = sexuality_romance_set
			}
		}
		
		set_character_flag = sexuality_romance_set
		
		if = { # Unlikely to be the case, but we might as well extend the effect...
			limit = {
				OR = {
					has_any_romance_interest_trait_trigger = yes
					has_character_flag = aro_ace
				}
				has_any_sex_interest_trait_trigger = no
				has_any_sexuality_trait_trigger = no
			}
			
			add_trait = asexual
		}
		
		else_if = { # For characters we don't want to lock in a specific sexuality on but have pre-defined some sexual interest for
			limit = {
				has_any_sex_interest_trait_trigger = yes
				has_any_sexuality_trait_trigger = no
			}
			
			randomize_valid_sexuality_effect = yes
			randomize_valid_romance_effect = yes
		}
		
		else_if = { # Usual case for anyone we pre-define a specific sexuality for
			limit = {
				has_any_sexuality_trait_trigger = yes
			}
			
			hidden_tooltip = {
				trigger_switch = {
					on_trigger = trait
					heterosexual = {
						if = {
							limit = {
								is_female = yes
							}
							add_trait = sex_interest_male
							if = {
								limit = {
									is_adult = yes
								}
								add_trait = sex_attraction_male
							}
						}
						else = {
							add_trait = sex_interest_female
							if = {
								limit = {
									is_adult = yes
								}
								add_trait = sex_attraction_female
							}
						}
					}
					homosexual = {
						if = {
							limit = {
								is_female = yes
							}
							add_trait = sex_interest_female
							if = {
								limit = {
									is_adult = yes
								}
								add_trait = sex_attraction_female
							}
						}
						else = {
							add_trait = sex_interest_male
							if = {
								limit = {
									is_adult = yes
								}
								add_trait = sex_attraction_male
							}
						}
					}
					bisexual = {
						add_trait = sex_interest_male
						add_trait = sex_interest_female
						if = {
							limit = {
								is_adult = yes
							}
							add_trait = sex_attraction_female
							add_trait = sex_attraction_male
						}
					}
					asexual = {
						if = {
							limit = {
								NOT = {
									has_character_flag = aro_ace # TIANXIATODO: Maybe a dedicated flag at some point?
								}
							}
							
							random_list = {
								25 = {
									if = {
										limit = {
											is_female = yes
										}
										add_trait = sex_interest_male
									}
									else = {
										add_trait = sex_interest_female
									}
								}
								25 = {
									if = {
										limit = {
											is_female = yes
										}
										add_trait = sex_interest_female
									}
									else = {
										add_trait = sex_interest_male
									}
								}
								25 = {
									add_trait = sex_interest_female
									add_trait = sex_interest_male
								}
								100 = {
									# No interest
								}
							}
						}
					}
				}
			}
			randomize_valid_romance_effect = yes
		}
		
		else = { # Nothing pre-defined
			randomize_sexuality_romance_effect = yes
		}
		
		# TIANXIATODO: Check if the character has aged past the attraction cutoff
		
		add_all_relevant_attraction_giving_traits_effect = yes
		add_all_relevant_attraction_giving_modifiers_effect = yes
	}
}

set_up_sexuality_romance_child_effect = {
	if = {
		limit = {
			NOT = {
				has_character_flag = sexuality_romance_set
			}
		}
		
		set_character_flag = sexuality_romance_set
		
		if = {
			limit = {
				NOT = {
					has_character_flag = aro_ace
				}
				has_any_sex_interest_trait_trigger = no
				has_any_sexuality_trait_trigger = no
			}
			
			# Full randomization
			on_birth_sexuality_romance_randomization_effect = yes
		}
		
		else_if = {
			limit = {
				NAND = {
					has_character_flag = aro_ace
					OR = {
						has_any_sex_interest_trait_trigger = yes
						has_character_flag = aro_ace
					}
					has_any_sexuality_trait_trigger = yes
				}
			}
			
			# Something was pre-defined
			
			if = {
				limit = {
					NAND = {
						has_any_sex_interest_trait_trigger = yes
						has_any_sexuality_trait_trigger = yes
					}
				}
				
				if = {
					limit = {
						has_any_sex_interest_trait_trigger = no
						has_any_sexuality_trait_trigger = no
					}
					
					# Randomize sexuality; nothing pre-defined
					random_list = {
						75 = {
							set_character_flag = is_heterosexual
						}
						10 = {
							set_character_flag = is_homosexual
						}
						10 = {
							set_character_flag = is_bisexual
						}
						5 = {
							set_character_flag = is_asexual
							random_list = {
								25 = {
									set_character_flag = asexual_opposite
								}
								25 = {
									set_character_flag = asexual_same
								}
								25 = {
									set_character_flag = asexual_both
								}
								100 = {
									set_character_flag = asexual_none
								}
							}
						}
					}
				}
				
				else_if = {
					limit = {
						has_any_sexuality_trait_trigger = yes
					}
					
					trigger_switch = {
						on_trigger = trait
						
						heterosexual = {
							if = {
								limit = {
									is_female = yes
								}
								add_trait = sex_interest_male
							}
							else = {
								add_trait = sex_interest_female
							}
						}
						
						homosexual = {
							if = {
								limit = {
									is_female = yes
								}
								add_trait = sex_interest_female
							}
							else = {
								add_trait = sex_interest_male
							}
						}
						
						bisexual = {
							add_trait = sex_interest_female
							add_trait = sex_interest_male
						}
						
						asexual = {
							if = {
								limit = {
									has_character_flag = aro_ace
								}
								
								set_character_flag = asexual_none
							}
							
							else = {
								random_list = {
									25 = {
										set_character_flag = asexual_opposite
									}
									25 = {
										set_character_flag = asexual_same
									}
									25 = {
										set_character_flag = asexual_both
									}
									100 = {
										set_character_flag = asexual_none
									}
								}
							}
						}
					}
				}
				
				else = {
					# Sex interest set, but not sexuality; randomize something valid
					if = {
						limit = {
							is_female = yes
						}
						random_list = {
							75 = {
								trigger = {
									NOT = {
										trait = sex_interest_female
									}
								}
								set_character_flag = is_heterosexual
							}
							10 = {
								trigger = {
									NOT = {
										trait = sex_interest_male
									}
								}
								set_character_flag = is_homosexual
							}
							10 = {
								set_character_flag = is_bisexual
							}
							5 = {
								trigger = {
									NOR = {
										trait = sex_attraction_female
										trait = sex_attraction_male
									}
								}
								set_character_flag = is_asexual
							}
						}
					}
					else = {
						random_list = {
							75 = {
								trigger = {
									NOT = {
										trait = sex_interest_male
									}
								}
								set_character_flag = is_heterosexual
							}
							10 = {
								trigger = {
									NOT = {
										trait = sex_interest_female
									}
								}
								set_character_flag = is_homosexual
							}
							10 = {
								set_character_flag = is_bisexual
							}
							5 = {
								trigger = {
									NOR = {
										trait = sex_attraction_female
										trait = sex_attraction_male
									}
								}
								set_character_flag = is_asexual
							}
						}
					}
				}
			}
			
			if = {
				limit = {
					NOT = {
						has_character_flag = aro_ace
					}
				}
				
				# Randomize romantic interest
				random_list = {
					25 = {
						trigger = {
							NOR = {
								has_character_flag = is_homosexual
								trait = homosexual
								AND = {
									is_female = yes
									trait = romance_attraction_female
								}
								AND = {
									is_female = no
									trait = romance_attraction_male
								}
							}
						}
						if = {
							limit = {
								is_female = yes
							}
							set_character_flag = romance_men
						}
						else = {
							set_character_flag = romance_women
						}
					}
					25 = {

on_adulthood_sexuality_romance_effect = {
	if = {
		limit = {
			has_any_sexuality_trait_trigger = no
		}
		trigger_switch = {
			on_trigger = has_character_flag
			
			is_heterosexual = { add_trait = heterosexual }
			is_homosexual = { add_trait = homosexual }
			is_bisexual = { add_trait = bisexual }
			is_asexual = { add_trait = asexual }
		}
		
		if = {
			limit = {
				trait = heterosexual
			}
			
			if = {
				limit = {
					is_female = yes
				}
				add_trait = sex_interest_male
			}
			else = {
				add_trait = sex_interest_female
			}
		}
		
		else_if = {
			limit = {
				trait = homosexual
			}
			
			if = {
				limit = {
					is_female = yes
				}
				add_trait = sex_interest_female
			}
			else = {
				add_trait = sex_interest_male
			}
		}
		
		else_if = {
			limit = {
				trait = bisexual
			}
			
			add_trait = sex_interest_female
			add_trait = sex_interest_male
		}
		
		else = {
			trigger_switch = {
				on_trigger = has_character_flag
				
				asexual_opposite = {
					if = {
						limit = {
							is_female = yes
						}
						add_trait = sex_interest_male
					}
					else = {
						add_trait = sex_interest_female
					}
				}
				
				asexual_same = {
					if = {
						limit = {
							is_female = yes
						}
						add_trait = sex_interest_female
					}
					else = {
						add_trait = sex_interest_male
					}
				}
				
				asexual_both = {
					add_trait = sex_interest_female
					add_trait = sex_interest_male
				}
			}
		}
	}
	
	if = {
		limit = {
			NOT = {
				trait = asexual
			}
		}
		if = {
			limit = {
				trait = sex_interest_female
			}
			add_trait = sex_attraction_female
		}
		if = {
			limit = {
				trait = sex_interest_male
			}
			add_trait = sex_attraction_male
		}
	}
	
	if = {
		limit = {
			has_any_romance_interest_trait_trigger = no
		}
		
		if = {
			limit = {
				has_character_flag = romance_women
			}
			
			add_trait = romance_interest_female
		}
		
		if = {
			limit = {
				has_character_flag = romance_men
			}
			
			add_trait = romance_interest_male
		}
