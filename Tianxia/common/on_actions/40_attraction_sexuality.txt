	}
}

	effect = {
		on_adulthood_sexuality_romance_effect = yes
	}
}

# Character scope
on_startup = {
	effect = {
		if = {
			limit = {
				is_adult = yes
			}
			
			set_up_sexuality_romance_effect = yes
		}
		else = {
			set_up_sexuality_romance_child_effect = yes
		}
	}
}

# Character scope
on_yearly_pulse = {
	effect = {
		if = {
			limit = {
				is_adult = yes
			}
			
			if = {
				limit = {
					practical_age >= 45
					is_actual_immortal_trigger = no
					NOT = {
						has_character_flag = attraction_cutoff
					}
				}
				attraction_cutoff_roll_effect = yes
			}
			
			remove_all_artifact_wonder_bloodline_focus_attraction_giving_modifiers = yes
			add_all_relevant_attraction_giving_modifiers_effect = yes
			
			update_fertility_penalty_effect = yes
		}
	}
}

# Character scope
on_new_holder = {
	effect = {
		# To catch randomly generated barons from e.g. the province decision
		if = {
			limit = {
				FROM = {
					tier = baron
				}
				NOT = {
					has_character_flag = sexuality_romance_set
				}
			}
			# Randomization gets wonky!
			change_variable = {
				which = global_random_ticking
				value = 1
			}
			# Keep the number of rerolls reasonable
			if = {
				limit = {
					check_variable = {
						which = global_random_ticking
						value >= 100
					}
				}
				modulo_variable = {
					which = global_random_ticking
					value = 100
				}
			}
			set_variable = {
				which = global_random_reroll
				which = global_random_ticking
			}
			while = {
				limit = {
					check_variable = {
						which = global_random_reroll
						value >= 0
					}
				}
				change_variable = {
					which = global_random_reroll
					value = -1
				}
				if = {
					limit = {
						is_variable_equal = {
							which = global_random_reroll
							value = -1
						}
					}
					if = {
						limit = {
							is_adult = yes
						}
						
						set_up_sexuality_romance_effect = yes
					}
					else = {
						set_up_sexuality_romance_child_effect = yes
					}
				}
			}
		}
	}
}
