on_chronicle_start = {
	effect = {
		# Ensure tribal levy building setup is carried out
		any_province = {
			limit = {
				has_owner = yes
				any_province_holding = {
					holding_type = tribal
				}
			}
			any_province_holding = {
				limit = {
					holding_type = tribal
				}
				if = {
					limit = {
						owner = {
							has_nomadic_tribal_government_trigger = no
						}
					}
					add_building = tb_base_tribal
				}
				else = {
					if = {
						limit = {
							owner = {
								has_steppe_nomadic_tribal_government_trigger = yes
							}
						}
						add_building = tb_base_nomad_steppe
					}
					else = {
						add_building = tb_base_nomad_desert
					}
				}
				refill_holding_levy = yes
			}
			if = {
				limit = {
					OR = {
						has_steppe_nomad_culture_trigger = yes
						has_desert_nomad_culture_trigger = yes
					}
					holder_scope = {
						OR = {
							trait = nomadic_trait
							is_tribal = no
						}
					}
				}
				add_province_modifier = {
					name = nomadic_land
					duration = -1
				}
			}
		}
		if = {
			limit = {
				has_game_rule = {
					name = merchant_republics
					value = tianxia
				}
			}
			any_playable_ruler = {
				limit = {
					is_patrician = yes
				}
				family_palace = {
					add_building = fp_inf_trade_posts
				}
			}
			if = {
				limit = {
					k_teutonic_state = {
						has_holder = yes
					}
					k_hansa = {
						has_holder = yes
					}
				}
				k_teutonic_state = {
					holder_scope = {
						any_realm_lord = {
							limit = {
								tier = baron
								government = republic_government
							}
							k_hansa = {
								holder_scope = {
									make_tributary = {
										who = PREVPREV
										tributary_type = trade_league_tributary
									}
								}
							}
						}
					}
				}
			}
		}
		if = {
			limit = {
				has_game_rule = {
					name = ere_bureaucratic
					value = yes
				}
			}
			e_byzantium = {
				holder_scope = {
					any_realm_lord = {
						limit = {
							is_government_potential = bureaucracy_government
						}
						set_government_type = bureaucracy_government
					}
				}
			}
			e_nicaea = {
				holder_scope = {
					any_realm_lord = {
						limit = {
							is_government_potential = bureaucracy_government
						}
						set_government_type = bureaucracy_government
					}
				}
			}
			e_trebizond = {
				holder_scope = {
					any_realm_lord = {
						limit = {
							is_government_potential = bureaucracy_government
						}
						set_government_type = bureaucracy_government
					}
				}
			}
			if = {
				limit = {
					is_feudal = yes
					independent = no
					top_liege = {
						government = roman_imperial_government
					}
					has_religion_features = no
					religion_group = christian
					OR = {
						culture = roman
						culture = sardinian
						culture = italian
						culture_group = byzantine
					}
				}
				set_government_type = bureaucracy_government
			}
		}
		else = {
			e_byzantium = {
				holder_scope = {
					any_realm_lord = {
						limit = {
							government = bureaucracy_government
						}
						set_government_type = feudal_government
					}
				}
			}
			e_nicaea = {
				holder_scope = {
					any_realm_lord = {
						limit = {
							government = bureaucracy_government
						}
						set_government_type = feudal_government
					}
				}
			}
			e_trebizond = {
				holder_scope = {
					any_realm_lord = {
						limit = {
							government = bureaucracy_government
						}
						set_government_type = feudal_government
					}
				}
			}
		}
		any_playable_ruler = {
			limit = {
				trait = nomadic_trait
				NOT = {
					any_demesne_province = {
						capital_holding = {
							holding_type = tribal
						}
					}
				}
			}
			remove_trait = nomadic_trait
			set_government_type = feudal_government # Will force validation
		}
	}
}
