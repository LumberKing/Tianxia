# The CB used by "Peasant revolter" characters
peasant_revolt = {
	name = CB_NAME_PEASANT_CONQUEST
	war_name = WAR_NAME_PEASANT_CONQUEST
	sprite = 16
	truce_days = 365

	hostile_against_others = yes
	check_all_titles = yes # if permanent, setting this to true will check against all of someones titles, including vassal held titles
	press_claim = yes
	attacker_can_call_allies = no
	defender_can_call_allies = no
	can_ask_to_join_war = no
	is_independence = yes

	can_use = {
		# Attacker
		ROOT = {
			primary_title = { temporary = yes } # Revolter or adventurer trigger
			war = no
			NOT = { is_vassal_or_below_of = FROM }
		}
	}

	can_use_title = {
		real_tier = COUNT
	}

	is_valid_title = {
		holder_scope = {
			OR = {
				character = FROM
				is_vassal_or_below_of = FROM
			}
		}
		NAND = {
			title = c_yamashiro
			ROOT = {
				gets_japanese_governments_trigger = yes
			}
			holder_scope = {
				OR = {
					is_the_tenno_trigger = yes
					any_liege = {
						is_the_tenno_trigger = yes
					}
				}
			}
		}
	}

	on_add = {
		fire_haruspicy_event_effect = yes
		anger_china_effect = yes
	}

	on_success_title = {
		usurp_title_plus_barony_if_unlanded = {
			target = attacker
			type = revolt
		}

		attacker = {
			show_scope_change = no

			hidden_effect = {
				# Trigger automatic government validation, so actual government will depend on capital holding
				if = {
					limit = { is_nomadic = yes }
					set_government_type = feudal_government
				}
			}
				
			hidden_tooltip = {
				if = {
					limit = {
						any_demesne_title = {
							tier = baron
							holding_type = castle
						}
					}
					
					# Should we have become a patrician by accident, fix that!
					any_demesne_title = {
						limit = {
							tier = baron
							is_titular = yes
						}
						
						unsafe_destroy_landed_title = THIS
					}
					
					set_correct_feudal_government_type_effect = yes # Fix stupid MRs!
				}
			}

			set_defacto_liege = attacker
		}

		defender = {
			show_scope_change = no

			prestige = -100
		}

		hidden_effect = {
			check_if_reconquista_finished_effect = yes

			defender = {
				landless_pope_effect = yes
			}
		}
		
		defender = {
			if = {
				limit = {
					government = chinese_imperial_government
				}
				mandate_penalty_small_effect = yes
			}
		}
	}

	on_fail_title = {
		attacker = {
			show_scope_change = no

			clear_wealth = yes
			imprison = defender

			hidden_effect = {
				add_character_modifier = {
					name = broken_spirit
					duration = -1
				}
			}
		}

		defender = {
			show_scope_change = no

			prestige = 10

			hidden_effect = {
				opinion = {
					name = opinion_rebel_traitor
					who = attacker
					years = 100
				}
			}
		}
		
		defender = {
			if = {
				limit = {
					government = chinese_imperial_government
				}
				mandate_boost_trivial_effect = yes
			}
		}
	}

	on_reverse_demand = {
		attacker = {
			show_scope_change = no

			clear_wealth = yes
			imprison = defender

			hidden_effect = {
				add_character_modifier = {
					name = broken_spirit
					days = -1
				}
			}
			
			if = {
				limit = {
					defender = {
						ai = yes
					}
				}
				
				any_courtier = {
					limit = {
						is_landed = no # Not anyone imprisoned
						trait = peasant_leader
					}
					
					if = {
						limit = {
							defender = {
								has_chinese_government_trigger = yes
							}
						}
						
						death = {
							death_reason = death_execution_thousand_cuts
							killer = defender
						}
					}
					
					else_if = {
						limit = {
							defender = {
								culture = roman
								NOT = {
									religion_group = christian
								}
							}
						}
						
						death = {
							death_reason = death_execution_crucifixion # First-time offence!
							killer = defender
						}
					}
					
					else = {
						death = {
							death_reason = death_execution_hdq
							killer = defender
						}
					}
				}
				
				if = {
					limit = {
						defender = {
							has_chinese_government_trigger = yes
						}
					}
					
					death = {
						death_reason = death_execution_thousand_cuts
						killer = defender
					}
				}
				
				else_if = {
					limit = {
						defender = {
							culture = roman
							NOT = {
								religion_group = christian
							}
						}
					}
					
					death = {
						death_reason = death_execution_crucifixion # First-time offence!
						killer = defender
					}
				}
				
				else = {
					death = {
						death_reason = death_execution_hdq
						killer = defender
					}
				}
			}
		}

		defender = {
			show_scope_change = no

			prestige = 20

			hidden_effect = {
				opinion = {
					name = opinion_rebel_traitor
					who = attacker
					years = 100
				}

				add_character_modifier = {
					name = crushed_revolt
					years = 10
				}
			}
		}
		
		defender = {
			if = {
				limit = {
					government = chinese_imperial_government
				}
				mandate_boost_small_effect = yes
			}
		}
	}

	attacker_ai_victory_worth = {
		factor = -1 # Always accept
	}

	attacker_ai_defeat_worth = {
		factor = 100
	}

	defender_ai_victory_worth = {
		factor = -1 # Always accept
	}

	defender_ai_defeat_worth = {
		factor = 100
	}
