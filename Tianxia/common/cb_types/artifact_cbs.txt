reclaim_heirloom_seal = {
	name = CB_NAME_RECLAIM_HEIRLOOM_SEAL
	war_name = WAR_NAME_RECLAIM_HEIRLOOM_SEAL
	sprite = 7
	truce_days = 3650
	sort_priority = 900
	
	is_permanent = yes
	can_ask_to_join_war = no
	allowed_to_target_tributaries = no
	defender_unoccupied_warscore = yes

	can_use_gui = {
		check_if_crusader_trigger = yes
		
		ROOT = {
			prestige >= 1000
		}
	}
	
	can_use = {
		ROOT = {
			OR = {
				claims_the_mandate_trigger = yes
				AND = {
					has_landed_title = e_china
					FROM = {
						claims_the_mandate_trigger = no
					}
				}
			}
		}
		FROM = {
			OR = {
				has_artifact = heirloom_seal_of_the_realm
				any_realm_character = {
					has_artifact = heirloom_seal_of_the_realm
				}
			}
			OR = {
				claims_the_mandate_trigger = no
				ROOT = {
					has_landed_title = e_china
				}
				has_character_modifier = mandate_of_heaven_1 # Lost
			}
		}
	}
	
	on_add = {
		fire_haruspicy_event_effect = yes
		anger_china_effect = yes
		
		attacker = {
			prestige = -1000
		}
		
		defender = {
			# TIANXIATODO: Chance to surrender early
		}
	}
	
	is_valid = {
		ROOT = {
			OR = {
				claims_the_mandate_trigger = yes
				AND = {
					has_landed_title = e_china
					FROM = {
						claims_the_mandate_trigger = no
					}
				}
			}
		}
		FROM = {
			OR = {
				has_artifact = heirloom_seal_of_the_realm
				any_realm_character = {
					has_artifact = heirloom_seal_of_the_realm
				}
			}
		}
	}
	
	on_success = {
		attacker = {
			if = {
				limit = {
					claims_the_mandate_trigger = yes
				}
				mandate_boost_massive_effect = yes
			}
			
			prestige = 1000
		}
		
		defender = {
			if = {
				limit = {
					claims_the_mandate_trigger = yes
				}
				mandate_penalty_massive_effect = yes
			}
			
			prestige = -1000
			
			custom_tooltip = {
				text = heirloom_seal_of_the_realm_transfer
				
				if = {
					limit = {
						has_artifact = heirloom_seal_of_the_realm
					}
					any_artifact = {
						limit = {
							artifact_type = heirloom_seal_of_the_realm
						}
						transfer_artifact = {
							from = PREV
							to = attacker
						}
					}
				}
				else = {
					any_realm_character = {
						limit = {
							has_artifact = heirloom_seal_of_the_realm
						}
						any_artifact = {
							limit = {
								artifact_type = heirloom_seal_of_the_realm
							}
							transfer_artifact = {
								from = PREV
								to = attacker
							}
						}
					}
				}
			}
		}
	}
	
	on_fail = {
		attacker = {
			if = {
				limit = {
					claims_the_mandate_trigger = yes
				}
				mandate_penalty_large_effect = yes
			}
			
			prestige = -500
		}
		
		defender = {
			if = {
				limit = {
					claims_the_mandate_trigger = yes
				}
				mandate_boost_large_effect = yes
			}
				
			prestige = 500
		}
	}
	
	on_reverse_demand = {
		attacker = {
			if = {
				limit = {
					claims_the_mandate_trigger = yes
				}
				mandate_penalty_massive_effect = yes
			}
			
			prestige = -1000

			transfer_scaled_wealth = {
				to = defender
				value = 5.0
			}
		}
		
		defender = {
			if = {
				limit = {
					claims_the_mandate_trigger = yes
				}
				mandate_boost_massive_effect = yes
			}
			
			prestige = 1000
		}
	}
}
