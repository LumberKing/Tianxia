		ROOT = {
			trigger_if = {
				limit = {
					FROM = {

			trigger_else_if = {

	}
	
	can_use = {
		has_game_rule = {
			name = merchant_republics
			value = tianxia
		}
		ROOT = {
			independent = yes
			is_merchant_republic = yes
			FROM = {
				any_realm_province = {
					port = yes
					ROOT = {
						any_trade_post = {
							is_coastal = yes
							can_naval_path_to = {
								target = PREVPREV
								distance < 500
							}
						}
					}
				}
			}
		}
		FROM = {
			independent = yes
			is_merchant_republic = no
			OR = {
				has_chinese_government_trigger = yes
				has_japanese_government_trigger = yes
				AND = {
					has_religion_feature = religion_dogmatic
					NOT = {
						religion = ROOT
					}
				}
			}
			NOT = {
				has_opinion_modifier = {
					who = ROOT
					name = forced_country_open_opinon
				}
			}
		}
	}
	
	on_add = {
		attacker = {
			show_scope_change = no
			
			if = {
				limit = {
					defender = {
						lower_real_tier_than = king
						controls_religion = no
					}
				}
				prestige = -200
			}
			else_if = {
				limit = {
					defender = {
						real_tier = king
						controls_religion = no
					}
				}
				prestige = -500
			}
			else = {
				prestige = -1000
			}
		}
		
		fire_haruspicy_event_effect = yes
		anger_china_effect = yes
	}
	
	is_valid = {
		ROOT = {
			independent = yes
			is_merchant_republic = yes
		}
	}
	
	on_success = {
		attacker = {
			if = {
				limit = {
					defender = {
						lower_real_tier_than = king
						controls_religion = no
					}
				}
				prestige = 100
			}
			else_if = {
				limit = {
					defender = {
						real_tier = king
						controls_religion = no

			}
		}

			if = {

					lower_real_tier_than = king
					controls_religion = no
				}
				prestige = -200
			}
			else_if = {
				limit = {
					real_tier = king
					controls_religion = no
				}
				prestige = -500
			}
			else = {
				prestige = -1000
			}
			
			reverse_remove_opinion = {
				who = attacker
				name = closed_the_country_opinion
			}
			opinion = {
				who = attacker
				name = forced_country_open_opinon
			}
			
			lift_embargo = attacker
		}
		defender = {
			if = {
				limit = {
					claims_the_mandate_trigger = yes
				}
				mandate_penalty_large_effect = yes
			}
		}
	}
	
	on_fail = {

			if = {

					defender = {
						lower_real_tier_than = king
						controls_religion = no
					}
				}
				prestige = -100
			}
			else_if = {
				limit = {
					defender = {
						real_tier = king
						controls_religion = no
					}
				}
				prestige = -250
			}
			else = {
				prestige = -500
			}
		}
		defender = {
			if = {

					lower_real_tier_than = king
