indian_reconquest_dharmic = {
	name = CB_NAME_INDIAN_RECONQUEST
	war_name = WAR_NAME_INDIAN_RECONQUEST
	sprite = 31
	truce_days = 3650
	
	is_permanent = yes
	can_ask_to_join_war = yes
	check_de_jure_tier = DUKE # this scans all dejure duchies for the counties which are held by or vassals(or below) of selected character. Only valid if is_permanent = yes
	allowed_to_target_tributaries = no
	
	can_use_gui = {
		ROOT = {
			china_forbids_war_trigger = no
			
			gets_chinese_governments_trigger = no
			gets_japanese_governments_trigger = no
		}
	}

	can_use = {
		ROOT = {
			has_landed_title = e_india
			gets_dharmic_feudal_trigger = yes
			is_feudal = yes
			
			custom_tooltip = {
				text = has_indian_bloodline
				any_owned_bloodline = {
					has_bloodline_flag = india_bloodline
					bloodline_is_active_for = PREV
				}
			}
			
			mercenary = no
			holy_order = no
		}
		
		FROM = {
			NOT = {
				same_realm = ROOT
			}
		}
	}

	can_use_title = {
		region = world_india
		
		any_direct_de_jure_vassal_title = { # Title must be inside holy war range (to keep pretenders from launching wars from a great distance)
			OR = {
				holder_scope = {
					OR = {
						character = ROOT
						is_liege_or_above = ROOT
					}
				}
				
				AND = {
					holder_scope = {
						OR = {
							character = FROM
							is_liege_or_above = FROM
						}
					}
					location = {
						any_neighbor_province = {
							OR = {
								AND = {
									has_owner = yes
									owner = {
										OR = {
											character = ROOT
											is_liege_or_above = ROOT
										}
									}
								}
								
								AND = {
									is_land = no
									any_neighbor_province = {
										OR = {
											AND = {
												has_owner = yes
												owner = {
													OR = {
														character = ROOT
														is_liege_or_above = ROOT
													}
												}
											}
											
											AND = {
												is_land = no
												any_neighbor_province = {
													owner = {
														OR = {
															character = ROOT
															is_liege_or_above = ROOT
														}
													}
												}
											}
										}
									}
								}
							}
						}
					}
				}
			}
		}
	}
	
	on_add = {
		fire_haruspicy_event_effect = yes
		anger_china_effect = yes
	}
	
	is_valid = {
		ROOT = {
			has_landed_title = e_india
			gets_dharmic_feudal_trigger = yes
			is_feudal = yes
			
			custom_tooltip = {
				text = has_indian_bloodline
				any_owned_bloodline = {
					has_bloodline_flag = india_bloodline
					bloodline_is_active_for = PREV
				}
			}
			NOT = {
				same_realm = FROM
			}
		}
	}
	
	on_success = {
		attacker = {
			participation_scaled_prestige = 100
		}
		
		any_attacker = {
			limit = { NOT = { character = attacker } }
			
			hidden_tooltip = { 
				participation_scaled_prestige = 100
			}
		}
		
		defender = { prestige = -100 }
		
		defender = {
			if = {
				limit = {
					claims_the_mandate_trigger = yes
				}
				mandate_penalty_medium_effect = yes
			}
		}
	}

	on_success_title = {
		custom_tooltip = { 
			text = other_invasion_succ_tip
			hidden_tooltip = {
				attacker = {
					vassalize_or_take_under_title = {
						title = PREV
						enemy = defender
						type = invasion
					}
				}
			}
		}
		
		if = {
			limit = {
				holder_scope = {
					character = defender
				}
			}
			
			usurp_title_only = {
				target = attacker
				type = invasion
			}
		}
	}

	on_fail = {
		attacker = {
			prestige = -150
		}
		
		defender = {
			if = {
				limit = {
					claims_the_mandate_trigger = yes
				}
				mandate_boost_small_effect = yes
			}
		}
	}

	on_reverse_demand = {
		attacker = {
			transfer_scaled_wealth = {
				to = FROM
				value = 4.0
			}
			
			prestige = -200
		}
		
		defender = {
			participation_scaled_piety = 100
			participation_scaled_prestige = 200
			
			if = {
				limit = {
					uses_decadence = yes
				}
				participation_scaled_decadence = -10
			}
		}
		
		any_defender = {
			limit = { NOT = { character = defender } }
			
			hidden_tooltip = { 
				participation_scaled_piety = 100
				participation_scaled_prestige = 200
				
				if = {
					limit = {
						uses_decadence = yes
					}
					participation_scaled_decadence = -10
				}
			}
		}
		
		defender = {
			if = {
				limit = {
					claims_the_mandate_trigger = yes
				}
				mandate_boost_medium_effect = yes
			}
		}
	}

	attacker_ai_victory_worth = {
		factor = -1 # always accept
	}
	
	attacker_ai_defeat_worth = {
		factor = 100
	}

	defender_ai_victory_worth = {
		factor = -1 # always accept
	}
	
	defender_ai_defeat_worth = {
		factor = 100
	}
}

								suzerain = ROOT

						}
					}

						any_neighbor_province = {
							OR = {
								AND = {
									has_owner = yes
									owner = {
										OR = {
											character = ROOT
											is_liege_or_above = ROOT
											is_tributary = {
												type = mandala_tributary
												suzerain = ROOT
											}
										}
									}
								}

										OR = {
											AND = {
												has_owner = yes
												owner = {

														is_liege_or_above = ROOT

													}

														OR = {
															character = ROOT
															is_liege_or_above = ROOT
															is_tributary = {
																type = mandala_tributary
																suzerain = ROOT
															}
														}
													}
												}
											}
										}
									}
								}
							}
						}
					}
				}
			}
		}
	}
	
	on_add = {
		fire_haruspicy_event_effect = yes
		anger_china_effect = yes
	}
	
	is_valid = {
		ROOT = {
			has_landed_title = e_india
			gets_mandala_government_trigger = yes
			is_feudal = yes
			
			custom_tooltip = {
				text = has_indian_bloodline
				any_owned_bloodline = {
					has_bloodline_flag = india_bloodline
					bloodline_is_active_for = PREV
				}
			}
			NOT = {
				same_realm = FROM
			}
		}
	}
	
	on_success = {

			participation_scaled_prestige = 100
		}
		
		any_attacker = {
			limit = { NOT = { character = attacker } }
			
			hidden_tooltip = { 
				participation_scaled_prestige = 100
			}
		}
		
		defender = { prestige = -100 }
		
		defender = {
			if = {
				limit = {
					claims_the_mandate_trigger = yes
				}
				mandate_penalty_medium_effect = yes
			}
		}
	}

	on_success_title = {
		save_event_target_as = target_duchy
		custom_tooltip = { 
			text = mandala_reconquest_tooltip
			defender = {
				if = {
					limit = {
						NOT = {
							any_realm_province = {
								county = {
									NOT = {
										de_jure_liege_or_above = event_target:target_duchy
									}
								}
							}
						}
					}
					
					attacker = {
						make_tributary = {
							who = PREV
							tributary_type = mandala_tributary
						}
					}
				}
				
				else = {
					any_realm_lord = {
						limit = {
							higher_real_tier_than = baron
							any_demesne_title = {
								de_jure_liege_or_above = event_target:mandala_subjugation_title
							}
							NOT = {
								any_demesne_title = {
									NOT = {
										de_jure_liege_or_above = event_target:mandala_subjugation_title
									}
								}
							}
						}
						
						primary_title = {
							set_defacto_liege = PREV
						}
						
						attacker = {
							make_tributary = {
								who = PREV
								tributary_type = mandala_tributary
							}
						}
					}
					
					create_random_soldier = {
						dynasty = random
						random_traits = yes
						culture = defender
						religion = defender
					}
						
					new_character = {
						vassalize_or_take_under_title = {
							title = event_target:target_duchy
							enemy = defender
						}
						
						primary_title = {
							set_defacto_liege = PREV
						}
						
						attacker = {
							make_tributary = {
								who = PREV
								tributary_type = mandala_tributary
							}
						}
					}
				}
			}
		}
		
		if = {
			limit = {
				holder_scope = {
					character = defender
					NOT = {
						suzerain = {
							character = attacker
						}
					}
				}
			}
			
			unsafe_destroy_landed_title = THIS
		}
	}

	on_fail = {
		attacker = {
			prestige = -150
		}
		
		defender = {
			if = {
				limit = {
					claims_the_mandate_trigger = yes
				}
				mandate_boost_small_effect = yes
			}
		}
	}

	on_reverse_demand = {
		attacker = {
			transfer_scaled_wealth = {
				to = FROM
				value = 4.0
			}
			
			prestige = -200
		}
		
		defender = {
			participation_scaled_piety = 100
			participation_scaled_prestige = 200
			
			if = {
				limit = {
					uses_decadence = yes
				}
				participation_scaled_decadence = -10
			}
		}
		
		any_defender = {
			limit = { NOT = { character = defender } }
			
			hidden_tooltip = { 
				participation_scaled_piety = 100
				participation_scaled_prestige = 200
				
				if = {
					limit = {
						uses_decadence = yes
					}
					participation_scaled_decadence = -10
				}
			}
		}
		
		defender = {
			if = {
				limit = {
					claims_the_mandate_trigger = yes
