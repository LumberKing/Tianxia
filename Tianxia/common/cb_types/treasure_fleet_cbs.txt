treasure_fleet_punitive_expedition = {
	name = CB_NAME_PUNITIVE_EXPEDITION
	war_name = WAR_NAME_PUNITIVE_EXPEDITION
	sprite = 30
	truce_days = 365
	is_permanent = no
	hostile_against_others = yes
	allow_whitepeace = no # You win, or you lose
	can_ask_to_join_war = no
	
	defender_unoccupied_warscore = yes
	
	is_valid = {
		e_china = {
			has_holder = yes
			holder_scope = {
				is_chinese_emperor_trigger = yes
			}
		}
		
		NOT = {
			has_global_flag = abort_treasure_fleet_asap
		}
		
		FROM = { # Defender
			independent = yes
		}
	}

	on_add = {
		fire_haruspicy_event_effect = yes
	}
	
	on_success = {
		attacker = {
			prestige = 1000
			
			disband_event_forces = tf_punitive_expedition
		}
		
		defender = {
			prestige = -1000
			
			hidden_tooltip = {
				if = {
					limit = {
						claims_the_mandate_trigger = yes
					}
					mandate_penalty_large_effect = yes
				}
			}
			
			e_china = {
				holder_scope = {
					reverse_imprison = FROM
					make_tributary = {
						who = FROM
						tributary_type = chinese_client_state
					}
					
					hidden_tooltip = {
						mandate_boost_large_effect = yes
						
						attacker = {
							move_character = PREV
							force_host = PREV
						}
					}
				}
			}
		}
	}
	
	on_reverse_demand = {
		attacker = {
			prestige = -1000
			
			disband_event_forces = tf_punitive_expedition
			
			imprison = defender
		}
		
		defender = {
			prestige = 1000
			roman_victory_triumph_effect = yes
		}
		
		hidden_tooltip = {
			e_china = {
				holder_scope = {
					mandate_penalty_large_effect = yes
				}
			}
			
			defender = {
				hidden_tooltip = {
					if = {
						limit = {
							claims_the_mandate_trigger = yes
						}
						mandate_boost_large_effect = yes
					}
				}
			}
		}
	}
	
	on_attacker_leader_death = {
		end_war = invalid
	}

	attacker_ai_victory_worth = {
		factor = -1 # always accept
	}
	
	attacker_ai_defeat_worth = {
		factor = 100
	}

	defender_ai_victory_worth = {
		factor = -1 # always accept
	}
	
	defender_ai_defeat_worth = {
		factor = 100
	}
}

treasure_fleet_invasion = {
	name = CB_NAME_TREASURE_FLEET_INVASION
	war_name = WAR_NAME_TREASURE_FLEET_INVASION
	sprite = 30
	truce_days = 365
	is_permanent = no
	hostile_against_others = yes
	allow_whitepeace = no # You win, or you lose
	attacker_can_call_allies = no # This might let vassals of China join, which would be stupid...
	can_ask_to_join_war = no
	check_de_jure_tier = KING
	
	defender_unoccupied_warscore = yes
	
	on_add = {
		fire_haruspicy_event_effect = yes
		
		set_global_flag = treasure_fleet_invasion_ongoing
	}
	
	is_valid = {
		ROOT = { # Attacker
			top_liege = {
				has_landed_title = e_china
			}
			
			independent = no # In case we become China...
		}
		
		FROM = { # Defender
			independent = yes
		}
	}
	
	on_success = {
		attacker = {
			clr_global_flag = treasure_fleet_invasion_ongoing
			
			prestige = 1000
			
			take_treasure_fleet_wealth_effect = yes
			
			remove_trait = treasure_fleet_voyage
			
			clear_flags_with_prefix = treasure_fleet_
		}
		
		defender = {
			prestige = -1000
			
			if = {
				limit = {
					claims_the_mandate_trigger = yes
				}
				mandate_penalty_massive_effect = yes
			}
		}
	}
	
	on_success_title = {
		attacker = {
			# 1: Get rid of old vassals
			any_vassal = {
				set_defacto_liege = e_china
			}
			
			# 2: Create e_treasure_fleet
			activate_title = {
				title = e_treasure_fleet
				status = yes
			}
			
			grant_title = e_treasure_fleet
			
			# 3: Flag all demesne titles except e_treasure_fleet
			hidden_tooltip = {
				any_demesne_title = {
					limit = {
						NOT = {
							title = e_treasure_fleet
						}
						is_titular = no
					}
					
					set_title_flag = give_to_china
				}
			}
			
			# 4: Invasion usupration
			defender = {
				any_realm_province = {
					limit = {
						has_province_flag = treasure_fleet_invasion_target_beachhead
					}
					
					county = {
						usurp_title = {
							target = attacker
							type = invasion
						}
					}
				}
			}
			
			custom_tooltip = {
				text = other_invasion_succ_tip
				hidden_tooltip = {
					attacker = {
						vassalize_or_take_under_title_destroy_duchies = {
							title = PREVPREV
							enemy = defender
							is_crusade = no
							type = invasion
						}
						
				
						if = {
							limit = {
								this_has_super_cool_nickname_trigger = no
								age < 25
								defender = {
									higher_tier_than = duke
								}
							}
							
							give_nickname = nick_the_little_conqueror
						}
						
						else_if = {
							limit = {
								can_apply_average_nickname_trigger = yes
								age >= 25
								defender = {
									higher_tier_than = duke
								}
							}
							
							give_nickname = nick_the_conqueror
						}
					}
				}
			}
			
			# 5: Give the EoC all flagged titles and remove the flags
			custom_tooltip = {
				text = GIVE_TITLES_TO_CHINA
				any_demesne_title = {
					limit = {
						has_title_flag = give_to_china
					}
					
					clr_title_flag = give_to_china
					
					e_china = {
						holder_scope = {
							grant_title_no_opinion = PREVPREV
						}
					}
				}
			}
			
			# 6: Imperial Warlord, if possible
			if = {
				limit = {
					gets_chinese_governments_trigger = yes
					NOT = {
						has_religion_feature = religion_gov_chinese
					}
				}
				
				custom_tooltip = {
					text = BECOME_WARLORD_CHINESE_IMPERIAL
					primary_title = {
						set_title_flag = pretender_chinese_empire
						
						add_law = {
							law = succ_primogeniture
							cooldown = no
							opinion_effect = no
						}
						
						if = {
							limit = {
								has_dlc = "Conclave"
							}
							add_law = {
								law = ze_administration_laws_2
								cooldown = no
								opinion_effect = no
							}
						}
						
						else = {
							add_law = {
								law = imperial_administration
								cooldown = no
								opinion_effect = no
							}
						}
						
						if = {
							limit = {
								has_dlc = "Charlemagne"
							}
							
							add_law = {
								law = vice_royalty_2
								cooldown = no
								opinion_effect = no
							}
						}
					}
					
					add_trait = warlord_trait
					
					set_government_type = warlord_chinese_imperial_government
					
					if = { # If we have positive Grace, zero it out!
						limit = {
							check_variable = {
								which = grace
								value = 1
							}
						}
						set_variable = {
							which = grace
							value = 0
						}
					}
					
					detract_grace_super_huge_effect = yes # China is VERY upset with you for doing this!
					
					detract_grace_super_huge_effect = yes # China is VERY upset with you for doing this!
					
					hidden_tooltip = {
						e_china = {
							holder_scope = {
								mandate_penalty_large_effect = yes
							}
						}
					}
				
					# Claim China
					e_china = {
						add_weak_pressed_claim = PREV
					}
					
					if = {
						limit = {
							has_character_modifier = peace_deal_with_china
						}
						
						remove_character_modifier = peace_deal_with_china
					}
					
					if = {
						limit = {
							has_character_modifier = chinese_imperial_trade_contract
						}
						
						remove_character_modifier = chinese_imperial_trade_contract
					}
					
					narrative_event = { id = soh.5505 } # Name the empire
				}
			}
			
			hidden_tooltip = {
				2753 = { # Hangzhou
					clean_up_all_treasure_fleet_flags_and_variables_effect = yes
				}
			}
			
			# Bloodline added here; we want to ensure ROOT is an emperor when it is given
			if = {
				limit = {
					is_female = no
				}
				
				create_bloodline = {
					type = treasure_fleet_invader_bloodline 
				}
			}
			
			else = {
				create_bloodline = {
					type = treasure_fleet_invader_bloodline 
					inheritance = matrilineal
				}
			}
			
			hidden_tooltip = {
				mandate_boost_massive_effect = yes
			}
			
			custom_tooltip = {
				text = new_capital_converts
				any_demesne_title = {
					limit = {
						location = {
							has_province_flag = treasure_fleet_invasion_target_beachhead
						}
					}
					
					culture = attacker
					religion = attacker
					
					location = {
						clr_province_flag = treasure_fleet_invasion_target_beachhead
					}
				}
			}
		}
		
		capital_scope = {
			county = {
				e_treasure_fleet = {
					set_preferred_capital = PREV
				}
			}
		}
	}
	
	on_reverse_demand = {
		attacker = {
			clr_global_flag = treasure_fleet_invasion_ongoing
			
			prestige = -1000
			
			disband_event_forces = treasure_fleet_invasion
			
			e_china = {
				holder_scope = {
					opinion = {
						who = attacker
						modifier = opinion_traitor
					}
					
					attacker = {
						set_defacto_liege = PREV
					
						imprison = PREV # Better than having the target do it...
					}
				}
			}
			
			any_demesne_title = {
				e_china = {
					holder_scope = {
						add_claim = PREVPREV
					}
				}
			}
			
			hidden_tooltip = {
				2753 = { # Hangzhou
					clean_up_all_treasure_fleet_flags_and_variables_effect = yes
				}
			}
			
			remove_trait = treasure_fleet_voyage
			
			clear_flags_with_prefix = treasure_fleet_
		}
		
		defender = {
			prestige = 1000
			
			roman_victory_triumph_effect = yes
			
			if = {
				limit = {
					has_dlc = "Jade Dragon"
					
					e_china = {
						holder_scope = {
							is_chinese_emperor_trigger = yes
						}
					}
					
					NOT = {
						claims_the_mandate_trigger = yes # Pretender emperors
					}
				}
				
				add_grace_super_huge_effect = yes
				
				add_grace_super_huge_effect = yes
			}
			
			hidden_tooltip = {
				any_independent_ruler = {
					any_realm_province = {
						limit = {
							has_province_flag = treasure_fleet_invasion_target_beachhead
						}
						
						clr_province_flag = treasure_fleet_invasion_target_beachhead
					}
				}
			}
			
			hidden_tooltip = {
				if = {
					limit = {
						claims_the_mandate_trigger = yes
					}
					mandate_boost_massive_effect = yes
				}
			}
		}
	}
	
	on_invalidation = {
		2753 = { # Hangzhou
			clr_global_flag = treasure_fleet_invasion_ongoing
			
			clean_up_all_treasure_fleet_flags_and_variables_effect = yes
		}
		
		attacker = {
			remove_trait = treasure_fleet_voyage
			
			hidden_tooltip = {
				any_independent_ruler = {
					any_realm_province = {
						limit = {
							has_province_flag = treasure_fleet_invasion_target_beachhead
						}
						
						clr_province_flag = treasure_fleet_invasion_target_beachhead
					}
				}
			}
		}
	}
	
	on_attacker_leader_death = {
		end_war = invalid
	}

	attacker_ai_victory_worth = {

	}
	
	attacker_ai_defeat_worth = {
