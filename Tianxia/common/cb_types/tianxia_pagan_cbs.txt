war_of_vengeance = {
	name = CB_NAME_WAR_OF_VENGEANCE
	war_name = WAR_NAME_WAR_OF_VENGEANCE
	sprite = 22
	truce_days = 365
	sort_priority = 3000

	hostile_against_others = no
	is_permanent = yes
	can_ask_to_join_war = yes
	check_de_jure_tier = DUKE # this scans all de jure kingdoms for the counties which are held by or vassals(or below) of selected character. Only valid if is_permanent = yes
	apply_short_occ_mod = no # Do not apply the 'recently_conquered' modifier to Holdings
	allowed_to_target_tributaries = no
	
	can_use_gui = {
		ROOT = {
			china_forbids_war_trigger = no
			
			OR = {
				AND = {
					piety >= 100
					OR = {
						has_religion_feature = religion_warlike
						has_religion_feature = religion_dogmatic
						AND = {
							religions_are_standard_trigger = yes
							OR = {
								religion = norse_pagan
								religion = tengri_pagan
								religion = aztec_pagan
								religion = zun_pagan
								religion = inca_pagan
								religion = jurchen_pagan
							}
						}
					}
				}
				piety >= 250
			}
			
			NOR = {
				gets_japanese_governments_trigger = yes
				AND = {
					gets_chinese_governments_trigger = yes
					is_tribal = no
					NOR = {
						government = successor_state_government
						is_any_warlord_trigger = yes
					}
				}
				any_liege = {
					OR = {
						gets_japanese_governments_trigger = yes
						AND = {
							gets_chinese_governments_trigger = yes
							is_tribal = no
							NOR = {
								government = successor_state_government
								is_any_warlord_trigger = yes
							}
						}
					}
				}
			}
			
			NOT = {
				any_liege = {
					NOT = {
						religion = ROOT
					}
				}
			}
			
			NOR = {
				trait = treasure_fleet_preparation
				trait = treasure_fleet_voyage
				trait = penglai_expedition_preparation
				trait = on_penglai_expedition
			}
			
			# No prestige cost here; it will be on the "Prepare invasion" decision
		}
	}
	
	can_use = {
		ROOT = {
			religion_group = pagan_group
			is_reformed_religion = no
			pacifist = no
		}
		FROM = {
			burned_ROOTs_tribal_pillar_trigger = yes
			NOR = {
				same_realm = ROOT
				religion = ROOT
			}
		}
	}
	
	can_use_title = {
		any_direct_de_jure_vassal_title = {
			holder_scope = {
				OR = {
					character = FROM
					is_vassal_or_below_of = FROM
				}
			}
			OR = {
				location = {
					religion = ROOT
					NOT = {
						has_province_modifier = nomadic_land
					}
				}
				is_holy_site = ROOT
				holder_scope = {
					religion = ROOT
				}
			}
			OR = {
				location = {
					any_neighbor_province = { # bordering province
						trigger_if = {
							limit = { has_owner = yes }

							owner = { # part of attacker's realm
								OR = {
									character = ROOT
									is_vassal_or_below_of = ROOT
								}
							}
						}
						
						trigger_else = {
							is_land = no # first sea zone
						
							trigger_if = {
								limit = {
									has_game_rule = {
										name = holy_war_distance
										value = technology
									}
								}
								
								ROOT = {
									capital_scope = {
										TECH_RECRUITMENT >= 2
										TECH_NAVAL >= 2
									}
								}
							}

							any_neighbor_province = {
								trigger_if = {
									limit = { has_owner = yes }

									owner = { # part of attacker's realm
										OR = {
											character = ROOT
											is_vassal_or_below_of = ROOT
										}
									}
								}
								trigger_else = {
									is_land = no # second sea zone
						
									trigger_if = {
										limit = {
											has_game_rule = {
												name = holy_war_distance
												value = technology
											}
										}
										
										ROOT = {
											capital_scope = {
												TECH_RECRUITMENT >= 4
												TECH_NAVAL >= 4
											}
										}
									}

									any_neighbor_province = {
										owner = { # part of attacker's realm
											OR = {
												character = ROOT
												is_vassal_or_below_of = ROOT
											}
										}
									}
								}
							}
						}
					}
				}
			}
		}
	}
	
	is_valid = {
		ROOT = {
			religion_group = pagan_group
			is_reformed_religion = no
		}
		FROM = {
			NOR = {
				same_realm = ROOT
				religion = ROOT
			}
		}
	}

	is_valid_title = {
		# Defender
		FROM = {
			num_of_realm_counties = {
				value >= 1
				title = PREV
			}
		}
		any_direct_de_jure_vassal_title = {
			holder_scope = {
				OR = {
					character = FROM
					is_vassal_or_below_of = FROM
				}
			}
			location = {
				religion = ROOT
			}
		}
	}
	
	on_add = {
		attacker = {
			show_scope_change = no
			
			if = {
				limit = {
					OR = {
						has_religion_feature = religion_warlike
						has_religion_feature = religion_dogmatic
						AND = {
							religions_are_standard_trigger = yes
							OR = {
								religion = norse_pagan
								religion = tengri_pagan
								religion = aztec_pagan
								religion = zun_pagan
								religion = inca_pagan
								religion = jurchen_pagan
							}
						}
					}
				}
				piety = -100
			}
			else = {
				piety = -250
			}
		}

		fire_haruspicy_event_effect = yes
		anger_china_effect = yes
	}
	
	on_success = {
		attacker = {
			show_scope_change = no

			participation_scaled_piety = 100
			participation_scaled_prestige = 50

			hidden_effect = {
				if = {
					limit = { has_ambition = obj_strengthen_religion }

					change_variable = {
						which = strengthen_religion
						value = 1
					}
				}

				if = {
					limit = {
						can_apply_cool_nickname_trigger = yes
					}

					random_list = {
						10 = {
							if = {
								limit = {
									is_female = no
								}
								give_nickname = nick_the_dragon
							}
							
							else = {
								give_nickname = nick_the_dragoness
							}
						}
						10 = {
							trigger = {
								can_apply_average_nickname_trigger = yes
							}
							give_nickname = nick_the_holy 
						}
						10 = {
							trigger = {
								can_apply_average_nickname_trigger = yes
							}
							give_nickname = nick_bloodaxe
						}
						10 = {
							trigger = {
								can_apply_average_nickname_trigger = yes
							}
							give_nickname = nick_the_great
						}
						40 = {
							give_nickname = nick_the_sword_of_god
						}
						
						100 = {
							trigger = {
								has_nickname = yes
							}
							# Nothing
						}
					}
				}
			}

			religion_authority = {
				name = won_holy_war
				years = 20
			}
		}

		any_attacker = {
			limit = { NOT = { character = attacker } }

			hidden_effect = {
				participation_scaled_piety = 100
				participation_scaled_prestige = 50

				if = {
					limit = { has_ambition = obj_strengthen_religion }

					change_variable = {
						which = strengthen_religion
						value = 1
					}
				}
			}
		}

		defender = {
			show_scope_change = no

			prestige = -50
			piety = -100

			religion_authority = {
				name = lost_holy_war
				years = 20
			}
		}
		
		attacker = {
			if = {
				limit = {
					government = chinese_imperial_government
				}
				mandate_boost_medium_effect = yes
			}
		}
		
		defender = {
			if = {
				limit = {
					government = chinese_imperial_government
				}
				mandate_penalty_medium_effect = yes
			}
		}
	}

	on_success_title = {
		custom_tooltip = {
			text = religious_cb_succ_tip_special
			
			any_direct_de_jure_vassal_title = {
				limit = {
					holder_scope = {
						OR = {
							character = FROM
							is_vassal_or_below_of = FROM
						}
					}
				}
				if = {
					limit = {
						holder_scope = {
							religion = attacker
						}
					}
					holder_scope = {
						set_defacto_liege = attacker
					}
				}
				
				else_if = {
					limit = {
						OR = {
							location = {
								religion = attacker
							}
							is_holy_site = attacker
						}
					}
					usurp_title_plus_barony_if_unlanded = {
						target = attacker
						type = holy_war
					}
				}
			}
		}
	}

	on_fail = {
		attacker = {
			show_scope_change = no

			piety = -50
			prestige = -100
		}
		
		defender = {
			if = {
				limit = {
					government = chinese_imperial_government
				}
				mandate_boost_small_effect = yes
			}
		}
		
		attacker = {
			if = {
				limit = {
					government = chinese_imperial_government
				}
				mandate_penalty_small_effect = yes
			}
		}
	}

	on_reverse_demand = {
		attacker = {
			show_scope_change = no

			transfer_scaled_wealth = {
				to = defender
				value = 4.0
			}

			religion_authority = {
				name = lost_holy_war
				years = 20
			}
		}

		defender = {
			show_scope_change = no

			participation_scaled_piety = 200
			participation_scaled_prestige = 100

			hidden_effect = {
				if = {
					limit = { has_ambition = obj_strengthen_religion }

					change_variable = {
						which = strengthen_religion
						value = 1
					}
				}
			}

			religion_authority = {
				name = won_holy_war
				years = 20
			}
		}

		any_defender = {
			limit = { NOT = { character = defender } }

			hidden_effect = {
				participation_scaled_piety = 200
				participation_scaled_prestige = 100

				if = {
					limit = { has_ambition = obj_strengthen_religion }

					change_variable = {
						which = strengthen_religion
						value = 1
					}
				}
			}
		}
		
		defender = {
			if = {
				limit = {
					government = chinese_imperial_government
				}
				mandate_boost_medium_effect = yes
			}
		}
		
		attacker = {
			if = {
				limit = {
					government = chinese_imperial_government
				}
				mandate_penalty_medium_effect = yes
			}
		}
	}

	attacker_ai_victory_worth = {
		factor = -1 # always accept
	}

	attacker_ai_defeat_worth = {
		factor = 100
	}

	defender_ai_victory_worth = {
		factor = -1 # always accept
	}

	defender_ai_defeat_worth = {
		factor = 100
	}
}

	allowed_to_target_tributaries = no

		# Attacker
		ROOT = {
			piety >= 1000
			
			NOT = {
				any_liege = {
					OR = {
						gets_japanese_governments_trigger = yes
						AND = {
							gets_chinese_governments_trigger = yes
							is_tribal = no
							NOR = {

			religion_group = pagan_group

			has_religion_feature = religion_dogmatic
			has_religion_feature = religion_temporal_head
			religion = FROM
			#is_nomadic = no
			mercenary = no
			NOT = { has_character_modifier = launched_subjugation }

			trigger_if = {
				limit = {
					ai = no
					multiplayer = yes
				}

				NOT = {
					has_game_rule = {
						name = multiplayer_invasions
						value = off
					}
				}
			}

			# Should use 'tribal_invasion' instead
			#trigger_if = {
			#	limit = { has_horde_culture = yes }
			#	NOT = { has_dlc = "Horse Lords" }
			#}
		}

		# Defender
		FROM = {
			NOT = {
				same_realm = ROOT
			}
			NOT = { is_offmap_governor = yes }
		}
	}

	can_use_title = {
		# Defender
		FROM = {
			num_of_realm_counties = {
				value >= 1
				title = PREV
			}
		}
		
		# The attacker needs at least one county in the target kingdom, or a border, or be only two sea zones away from one of your counties
		any_direct_de_jure_vassal_title = {
			any_direct_de_jure_vassal_title = {
				OR = {
					holder_scope = {
						OR = {
							character = ROOT
							is_liege_or_above = ROOT
						}
					}
					
					AND = {
						holder_scope = {
							OR = {
								character = FROM
								is_liege_or_above = FROM
							}
						}
						location = {
							any_neighbor_province = {
								OR = {
									AND = {
										has_owner = yes
										owner = {
											OR = {
												character = ROOT
												is_liege_or_above = ROOT
											}
										}
									}
									
									AND = {
										is_land = no
										any_neighbor_province = {
											OR = {
												AND = {
													has_owner = yes
													owner = {
														OR = {
															character = ROOT
															is_liege_or_above = ROOT
														}
													}
												}
												
												AND = {
													is_land = no
													any_neighbor_province = {
														owner = {
															OR = {
																character = ROOT
																is_liege_or_above = ROOT
															}
														}
													}
												}
											}
										}
									}
								}
							}
						}
					}
				}
			}
		}
	}
	
	is_valid = {
		ROOT = {
			has_dlc = "Holy Fury"
			controls_religion = yes
			religion_group = pagan_group
			government = holy_mandate_government
			has_religion_feature = religion_dogmatic
			religion = FROM
			#is_nomadic = no
			mercenary = no
			NOT = {
				any_liege = {
					OR = {
						gets_japanese_governments_trigger = yes
						AND = {
							gets_chinese_governments_trigger = yes
							is_tribal = no
							NOR = {
								government = successor_state_government
								is_any_warlord_trigger = yes
							}
						}
					}
				}
			}
		}
		
		FROM = {
			religion = ROOT
		}
	}

	is_valid_title = {
		# Defender
		FROM = {
			num_of_realm_counties = {
				value >= 1
				title = PREV
			}
		}
	}

	on_add = {
		attacker = {
			show_scope_change = no
			
			piety = -1000
			
			if = {
				limit = {
					controls_religion = yes
				}
				add_character_modifier = {
					name = launched_subjugation
					years = 10
					hidden = yes
				}
			}
			else = {
				add_character_modifier = {
					name = launched_subjugation
					years = 25
					hidden = yes
				}
			}
		}

		fire_haruspicy_event_effect = yes
		anger_china_effect = yes
	}

	on_success = {
		attacker = {
			show_scope_change = no

			prestige = 100
			participation_scaled_prestige = 100
		}

		any_attacker = {
			limit = { NOT = { character = attacker } }
			hidden_effect = { participation_scaled_prestige = 100 }
		}

		defender = {
			show_scope_change = no

			prestige = -200
		}
		
		attacker = {
			if = {
				limit = {
					claims_the_mandate_trigger = yes
				}
				mandate_boost_large_effect = yes
			}
		}
		
		defender = {
			if = {
				limit = {
					claims_the_mandate_trigger = yes
				}
				mandate_penalty_large_effect = yes
			}
		}
	}

	on_success_title = {
		custom_tooltip = {
			text = pagan_subjugation_tip

			attacker = {
				subjugate_or_take_under_title = { # If the target only has territory within the kingdom, he is simply vassalized
					title = PREV
					enemy = defender
				}
			}
		}
	}

	on_fail = {
		attacker = {
			show_scope_change = no

			prestige = -100
		}

		defender = {
			show_scope_change = no

			prestige = 50
			participation_scaled_prestige = 50
		}

		any_defender = {
			limit = { NOT = { character = defender } }
			hidden_effect = { participation_scaled_prestige = 50 }
		}
		
		defender = {
			if = {
				limit = {
					claims_the_mandate_trigger = yes
				}
				mandate_boost_medium_effect = yes
			}
		}
		
		attacker = {
			if = {
				limit = {
					claims_the_mandate_trigger = yes
				}
				mandate_penalty_medium_effect = yes
			}
		}
	}

	on_reverse_demand = {
		attacker = {
			show_scope_change = no

			prestige = -200

			transfer_scaled_wealth = {
				to = defender
				value = 2.0
			}
		}

		defender = {
			show_scope_change = no

			prestige = 100
			participation_scaled_prestige = 100
			roman_victory_triumph_effect = yes
		}

		any_defender = {
			limit = { NOT = { character = defender } }
			hidden_effect = { participation_scaled_prestige = 100 }
		}
		
		defender = {
			if = {
				limit = {
					claims_the_mandate_trigger = yes
				}
				mandate_boost_large_effect = yes
			}
		}
		
		attacker = {
			if = {
				limit = {
					claims_the_mandate_trigger = yes
				}
				mandate_penalty_large_effect = yes
			}
		}
	}

	attacker_ai_victory_worth = {
		factor = -1 # always accept
	}

	attacker_ai_defeat_worth = {
		factor = 100
	}

	defender_ai_victory_worth = {
		factor = -1 # always accept
	}

	defender_ai_defeat_worth = {
		factor = 100
	}

	ai_will_do = {
		factor = 0.9 # Prefer using CB's without on_add costs
	}
}
