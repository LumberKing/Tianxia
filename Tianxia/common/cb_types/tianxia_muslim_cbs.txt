free_caliph_cb = {

	war_name = WAR_NAME_FREE_CALIPH
	sprite = 10
	truce_days = 3650

	sort_priority = 1000
	
	is_permanent = yes
	can_ask_to_join_war = yes

		ROOT = {

			china_forbids_war_trigger = no

		FROM = {
			independent = yes
			any_realm_lord = {
				controls_religion = yes

			}
		}
	}
	
	on_add = {
		attacker = {
			show_scope_change = no
			
			piety = -500
		}

		fire_haruspicy_event_effect = yes
		anger_china_effect = yes
	}
	
	on_success = {
		attacker = {
			piety = 500
		}
		defender = {
			if = {
				limit = {
					religion_group = muslim
					NOR = {
						religion = yazidi
						religion = druze
					}
				}
				piety = -500
			}
			else = {
				prestige = -1000
			}
			any_realm_lord = {
				limit = {
					controls_religion = yes
					religion = ROOT
				}
				set_defacto_liege = THIS
				any_demesne_title = {
					limit = {
						controls_religion = yes
						claimed_by = FROM
					}
					remove_claim = FROM
				}
			}
		}
		attacker = {
			if = {
				limit = {
					claims_the_mandate_trigger = yes
				}
				mandate_boost_massive_effect = yes
			}
		}
		
		defender = {
			if = {
				limit = {
					claims_the_mandate_trigger = yes
				}
				mandate_penalty_massive_effect = yes
			}
		}
	}
	
	on_fail = {
		attacker = {
			piety = -250
		}
		defender = {
			if = {
				limit = {
					religion_group = muslim
					NOR = {
						religion = yazidi
						religion = druze
					}
				}
				piety = 250
			}
			else = {
				prestige = 500
			}
		}
		defender = {
			if = {
				limit = {
					claims_the_mandate_trigger = yes
				}
				mandate_boost_large_effect = yes
			}
		}
		
		attacker = {
			if = {
				limit = {
					claims_the_mandate_trigger = yes
				}
				mandate_penalty_large_effect = yes
			}
		}
	}
	
	on_reverse_demand = {
		attacker = {
			piety = -500

			transfer_scaled_wealth = {
				to = defender
				value = 4.0
			}
		}
		defender = {
			if = {
				limit = {
					religion_group = muslim
					NOR = {
						religion = yazidi
						religion = druze
					}

			if = {
				limit = {
					claims_the_mandate_trigger = yes
				}
				mandate_penalty_massive_effect = yes
			}
		}
	}

	attacker_ai_victory_worth = {
		factor = -1 # always accept
	}

	attacker_ai_defeat_worth = {
		factor = 100
	}

	defender_ai_victory_worth = {
		factor = -1 # always accept
	}

	defender_ai_defeat_worth = {
		factor = 100
	}

	ai_will_do = {
		factor = 0.9 # Prefer using CB's without on_add costs
		
		trigger = { # Focus on your claims!
			FROM = {
				NOT = {
					any_demesne_title = {
						higher_tier_than = duke
						claimed_by = ROOT
					}
				}
			}
		}
	}
}
