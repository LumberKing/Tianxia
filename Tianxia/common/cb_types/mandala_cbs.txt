mandala_capital_duchy_subjugation = {
	name = CB_NAME_PAGAN_SUBJUGATION
	war_name = WAR_NAME_PAGAN_SUBJUGATION
	sprite = 8
	truce_days = 365
	sort_priority = 1000

	hostile_against_others = yes
	is_permanent = yes
	can_ask_to_join_war = yes
	check_de_jure_tier = DUKE # this scans all de jure kingdoms for the counties which are held by or vassals(or below) of selected character. Only valid if is_permanent = yes
	allowed_to_target_tributaries = no

	can_use_gui = {
		# Attacker
		ROOT = {
			piety >= 100
			
			china_forbids_war_trigger = no
			
			NOR = {
				gets_japanese_governments_trigger = yes
				AND = {
					gets_chinese_governments_trigger = yes
					is_tribal = no
					NOR = {
						government = successor_state_government
						is_any_warlord_trigger = yes
					}
				}
				any_liege = {
					OR = {
						gets_japanese_governments_trigger = yes
						AND = {
							gets_chinese_governments_trigger = yes
							is_tribal = no
							NOR = {
								government = successor_state_government
								is_any_warlord_trigger = yes
							}
						}
					}
				}
			}
			
			NOR = {
				trait = treasure_fleet_preparation
				trait = treasure_fleet_voyage
				trait = penglai_expedition_preparation
				trait = on_penglai_expedition
			}
		}
	}

	can_use = {
		# Attacker
		ROOT = {
			#has_dharmic_religion_trigger = yes
			culture_group = FROM
			NOT = { same_realm = FROM }
			#has_horde_culture = no # Should use tribal invasion instead
			is_nomadic = no
			mercenary = no
			gets_mandala_government_trigger = yes
			NOR = {
				has_nomadic_tribal_government_trigger = yes
				government = successor_state_government
			}
			
			NOT = {
				has_character_modifier = launched_subjugation
			}
		}

		# Defender
		FROM = {
			OR = {
				gets_dharmic_feudal_trigger = yes
				gets_mandala_government_trigger = yes
			}
			NOR = {
				has_nomadic_tribal_government_trigger = yes
				government = successor_state_government
			}

			NOR = {
				is_offmap_governor = yes

				any_war = {
					defender = { character = PREVPREV }
					using_any_subjugation_cb_trigger = yes
				}
				
				any_realm_lord = {
					character = ROOT
				}
			}
		}
	}

	can_use_title = {
		# Attacker
		ROOT = {
			capital_scope = {
				duchy = {
					title = PREVPREVPREV
				}
			}
		}

		# Defender
		FROM = {
			num_of_realm_counties = {
				value >= 1
				title = PREV
			}
		}
	}
	
	is_valid = {
		ROOT = {
			gets_dharmic_feudal_trigger = yes
		}
		
		FROM = {
			OR = {
				gets_dharmic_feudal_trigger = yes
				gets_mandala_government_trigger = yes
			}
		}
	}

	is_valid_title = {
		# Defender
		FROM = {
			num_of_realm_counties = {
				value >= 1
				title = PREV
			}
		}
	}

	on_add = {
		attacker = {
			show_scope_change = no
			piety = -100

			add_character_modifier = {
				name = launched_subjugation
				years = 10
				hidden = yes
			}
		}

		fire_haruspicy_event_effect = yes
		anger_china_effect = yes
	}

	on_success = {
		attacker = {
			show_scope_change = no

			participation_scaled_prestige = 200
		}

		any_attacker = {
			limit = { NOT = { character = attacker } }
			hidden_effect = { participation_scaled_prestige = 200 }
		}

		defender = {
			show_scope_change = no

			prestige = -200
		}
		
		attacker = {
			if = {
				limit = {
					claims_the_mandate_trigger = yes
				}
				mandate_boost_large_effect = yes
			}
		}
		
		defender = {
			if = {
				limit = {
					claims_the_mandate_trigger = yes
				}
				mandate_penalty_large_effect = yes
			}
		}
	}

	on_success_title = {
		# Usurp the target kingdom itself, if the enemy holds it
		if = {
			limit = {
				holder = defender

				attacker = {
					lower_real_tier_than = EMPEROR
				}
			}

			attacker = {
				show_scope_change = no

				usurp_title = {
					target = PREV
					type = invasion
				}
			}
		}

		if = {
			# Take the enemy's duchies within the target kingdom, if we are the same or lower tier
			limit = {
				attacker = {
					NOT = { higher_real_tier_than = defender }
				}
			}

			defender = {
				show_scope_change = no

				any_demesne_title = {
					limit = {
						real_tier = DUKE

						OR = {
							title = PREVPREV
							de_jure_liege_or_above = PREVPREV
						}
					}

					attacker = {
						show_scope_change = no

						usurp_title = {
							target = PREV
							type = invasion
						}
					}
				}
			}
		}

		# Have I become the vassal of the loser due to taking a duchy or kingdom?
		# TODO: Ensure vassals using this CB can't become independent!
		if = {
			limit = {
				attacker = {
					is_vassal_or_below_of = defender
				}
			}

			attacker = {
				set_defacto_liege = THIS
			}
		}

		# Vassalize the rest
		custom_tooltip = {
			text = pagan_subjugation_tip

			attacker = {
				subjugate_or_take_under_title = { # If the target only has territory within the kingdom, he is simply vassalized
					title = PREV
					enemy = defender
				}
			}
		}
	}

	on_fail = {
		attacker = {
			show_scope_change = no

			prestige = -50
		}

		defender = {
			show_scope_change = no

			participation_scaled_prestige = 100
		}

		any_defender = {
			limit = { NOT = { character = defender } }
			hidden_effect = { participation_scaled_prestige = 100 }
		}
		
		defender = {
			if = {
				limit = {
					claims_the_mandate_trigger = yes
				}
				mandate_boost_medium_effect = yes
			}
		}
		
		attacker = {
			if = {
				limit = {
					claims_the_mandate_trigger = yes
				}
				mandate_penalty_medium_effect = yes
			}
		}
	}

	on_reverse_demand = {
		attacker = {
			show_scope_change = no

			prestige = -100

			transfer_scaled_wealth = {
				to = defender
				value = 2.0
			}
		}

		defender = {
			show_scope_change = no

			participation_scaled_prestige = 200
		}

		any_defender = {
			limit = { NOT = { character = defender } }
			hidden_effect = { participation_scaled_prestige = 200 }
		}
		
		defender = {
			if = {
				limit = {
					claims_the_mandate_trigger = yes
				}
				mandate_boost_large_effect = yes
			}
		}
		
		attacker = {
			if = {
				limit = {
					claims_the_mandate_trigger = yes
				}
				mandate_penalty_large_effect = yes
			}
		}
	}

	on_invalidation = {
		attacker = {
			show_scope_change = no

			remove_character_modifier = launched_subjugation
			piety = 100
		}
	}

	attacker_ai_victory_worth = {
		factor = -1 # always accept
	}

	attacker_ai_defeat_worth = {
		factor = 100
	}

	defender_ai_victory_worth = {
		factor = -1 # always accept
	}

	defender_ai_defeat_worth = {
		factor = 100
	}

	ai_will_do = {
		factor = 0.9 # Prefer using CB's without on_add costs

		trigger = {
			trigger_if = {
				limit = {
					ROOT = {
						is_tribal = yes
					}

					FROM = {
						is_nomadic = yes
					}
				}

				any_direct_de_jure_vassal_title = {
					any_direct_de_jure_vassal_title = {
						OR = {
							holding_type = castle
							holding_type = city
							holding_type = tribal
						}
					}
				}
			}
			trigger_else_if = {
				limit = {
					ROOT = {
						is_nomadic = no
						is_tribal = no
					}

					FROM = {
						is_nomadic = yes
					}

				}

				any_direct_de_jure_vassal_title = {
					any_direct_de_jure_vassal_title = {
						OR = {
							holding_type = castle
							holding_type = city
						}
					}
				}
			}

			#trigger_if = {
			#	limit = {
			#		ROOT = {
			#			has_character_flag = is_temujin
			#			has_landed_title = e_mongol_empire
			#			is_landed = yes
			#		}
			#	}
			#
			#	FROM = {
			#		NOT = { has_landed_title = e_china_west_governor }
			#	}
			#}
		}
	}
}

			num_of_realm_counties = {
				value >= 1

		}
	}

	on_add = {
		attacker = {
			show_scope_change = no
			
			if = {
				limit = {
					defender = {
						government = dharmic_feudal_government
					}
				}
				piety = -2000
			}
			else_if = {
				limit = {
					defender = {
						NOT = {
							culture_group = ROOT
						}
					}
				}
				piety = -1000
			}
			else = {
				piety = -500
			}

			add_character_modifier = {
				name = launched_subjugation
				years = 25
				hidden = yes
			}
		}

		fire_haruspicy_event_effect = yes
		anger_china_effect = yes
	}

	on_success = {
		attacker = {
			show_scope_change = no

			participation_scaled_prestige = 200
			
			custom_tooltip = {
				text = mandala_subjugation_victory
			}
		}

		any_attacker = {
			limit = { NOT = { character = attacker } }
			hidden_effect = { participation_scaled_prestige = 200 }
		}

		defender = {
			show_scope_change = no

			prestige = -200
		}
		
		attacker = {
			if = {
				limit = {
					claims_the_mandate_trigger = yes
				}
				mandate_boost_large_effect = yes
			}
		}
		
		defender = {
			if = {
				limit = {
					claims_the_mandate_trigger = yes
				}
				mandate_penalty_large_effect = yes
			}
		}
	}

	on_success_title = {
		# Destroy the target kingdom itself, if the enemy holds it and should lose it
		save_event_target_as = mandala_subjugation_title
		
		if = {
			limit = {
				holder = defender
			}
			
			defender = {
				any_realm_lord = {
					limit = {
						higher_real_tier_than = baron
						any_demesne_title = {
							de_jure_liege_or_above = event_target:mandala_subjugation_title
						}
						NOT = {
							any_demesne_title = {
								NOT = {
									de_jure_liege_or_above = event_target:mandala_subjugation_title
								}
							}
						}
					}
					
					set_character_flag = mandala_tributary_@ROOT
					
					primary_title = {
						set_defacto_liege = PREV
					}
					
					set_defacto_liege = THIS
				}
			}
			
			if = {
				limit = {
					OR = {
						defender = {
							any_realm_province = {
								county = {
									NOT = {
										de_jure_liege_or_above = event_target:mandala_subjugation_title
									}
								}
							}
						}
						attacker = {
							lower_real_tier_than = king
						}
					}
				}
				unsafe_destroy_landed_title = THIS
				add_pressed_claim = defender
			}
			
			if = {
				limit = {
					defender = {
						any_realm_province = {
							county = {
								NOT = {
									de_jure_liege_or_above = event_target:mandala_subjugation_title
								}
							}
						}
					}
				}
				
				defender = {
					custom_tooltip = {
						text = defender_loses_land_in_kingdom
						
						create_random_soldier = {
							dynasty = random
							random_traits = yes
							culture = defender
							religion = defender
						}
						
						new_character = {
							vassalize_or_take_under_title = {
								title = event_target:mandala_subjugation_title
								enemy = defender
							}
							
							primary_title = {
								set_defacto_liege = PREV
							}
							
							set_defacto_liege = THIS
							
							attacker = {
								make_tributary = {
									who = PREV
									tributary_type = mandala_tributary
								}
							}
						}
					}
				}
			}
			
			else = {
				defender = {
					if = {
						limit = { is_tributary = yes }
						remove_tributary = suzerain
					}
					
					attacker = {
						make_tributary = {
							who = defender
							tributary_type = mandala_tributary
						}
					}
				}
			}
			
			custom_tooltip = {
				text = establish_tributaries_in_kingdom
				
				any_playable_ruler = {
					limit = {
						has_character_flag = mandala_tributary_@ROOT
					}
					
					any_tributary = {
						remove_tributary = suzerain
					}
					
					clr_character_flag = mandala_tributary_@ROOT
					
					if = {
						limit = { is_tributary = yes }
						remove_tributary = suzerain
					}
					
					attacker = {
						make_tributary = {
							who = PREV
							tributary_type = mandala_tributary
						}
					}
				}
			}
		}
		
		else = {
			defender = {
				any_realm_lord = {
					limit = {
						higher_real_tier_than = baron
						any_demesne_title = {
							de_jure_liege_or_above = event_target:mandala_subjugation_title
						}
						NOT = {
							any_demesne_title = {
								NOT = {
									de_jure_liege_or_above = event_target:mandala_subjugation_title
								}
							}
						}
					}
					
					set_character_flag = mandala_tributary_@ROOT
					
					primary_title = {
						set_defacto_liege = PREV
					}
					
					set_defacto_liege = THIS
				}
			}
			
			custom_tooltip = {
				text = establish_tributaries_in_kingdom
				
				any_playable_ruler = {
					limit = {
						has_character_flag = mandala_tributary_@ROOT
					}
					
					any_tributary = {
						remove_tributary = suzerain
					}
					
					clr_character_flag = mandala_tributary_@ROOT
					
					if = {
						limit = { is_tributary = yes }
						remove_tributary = suzerain
					}
					
					attacker = {
						make_tributary = {
							who = PREV
							tributary_type = mandala_tributary
						}
					}
				}
			}
			
			if = {
				limit = {
					defender = {
						any_realm_province = {
