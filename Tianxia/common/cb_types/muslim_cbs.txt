muslim_county_conquest = {
	name = CB_NAME_COUNTY_CONQUEST
	war_name = WAR_NAME_COUNTY_CONQUEST
	sprite = 16
	truce_days = 1825
	sort_priority = 700

	hostile_against_others = yes
	is_permanent = yes
	check_all_titles = yes # if permanent, setting this to true will check against all of someones titles, including vassal held titles
	can_ask_to_join_war = no
	allowed_to_target_tributaries = no

	can_use_gui = {
		# Attacker
		ROOT = {
			piety >= 50
			
			china_forbids_war_trigger = no
			
			NOR = {
				gets_japanese_governments_trigger = yes
				AND = {
					gets_chinese_governments_trigger = yes
					is_tribal = no
					NOR = {
						government = successor_state_government
						is_any_warlord_trigger = yes
					}
				}
				any_liege = {
					OR = {
						gets_japanese_governments_trigger = yes
						AND = {
							gets_chinese_governments_trigger = yes
							is_tribal = no
							NOR = {
								government = successor_state_government
								is_any_warlord_trigger = yes
							}
						}
					}
				}
			}
			
			NOR = {
				trait = treasure_fleet_preparation
				trait = treasure_fleet_voyage
				trait = penglai_expedition_preparation
				trait = on_penglai_expedition
			}
		}
	}

	can_use = {
		# Attacker
		ROOT = {
			religion_group = muslim
			NOR = {
				religion = yazidi
				religion = druze
			}
			NOT = { is_vassal_or_below_of = FROM }
			#is_nomadic = no # Use the 'pagan_county_conquest' CB instead
			OR = {
				is_tribal = yes
				government = ghazi_government
			}
			has_nomadic_tribal_government_trigger = no
			mercenary = no
		}
	}

	can_use_title = {
		real_tier = COUNT

		holder_scope = {
			OR = {
				character = FROM
				is_vassal_or_below_of = FROM
			}
		}

		# Only bordering counties
		location = {
			any_neighbor_province = {
				owner = {
					OR = {
						character = ROOT
						is_vassal_or_below_of = ROOT
					}
				}
			}
		}
	}
	
	is_valid = {
		ROOT = {
			religion_group = muslim
			NOR = {
				religion = yazidi
				religion = druze
			}
		}
	}

	is_valid_title = {
		holder_scope = {
			OR = {
				character = FROM
				is_vassal_or_below_of = FROM
			}
		}
	}

	on_add = {
		attacker = {
			show_scope_change = no

			piety = -50
		}

		fire_haruspicy_event_effect = yes
		anger_china_effect = yes
	}
	
	on_success = {
		attacker = {
			if = {
				limit = {
					claims_the_mandate_trigger = yes
				}
				mandate_boost_small_effect = yes
			}
		}
		
		defender = {
			if = {
				limit = {
					claims_the_mandate_trigger = yes
				}
				mandate_penalty_small_effect = yes
			}
		}
	}

	on_success_title = {
		if = { # Vassalise one county counts
			limit = {
				holder_scope = {
					real_tier = COUNT
					num_of_count_titles == 1
					lower_real_tier_than = attacker
					is_patrician = no
				}
			}

			holder_scope = {
				show_scope_change = no

				set_defacto_liege = attacker
			}
		}

		else = { # Usurp title under other circumstances
			usurp_title_plus_barony_if_unlanded = {
				target = attacker
				type = invasion
			}

			any_de_jure_vassal_title = { # take all baronies under the one we're fighting for
				show_scope_change = no

				limit = {
					NOT = { de_facto_liege = PREV }

					holder_scope = {
						OR = {
							character = defender
							is_vassal_or_below_of = defender
						}
					}
				}

				usurp_title_plus_barony_if_unlanded = {
					target = attacker
					type = invasion
				}
			}
		}

		attacker = {
			show_scope_change = no

			participation_scaled_prestige = 100
		}

		any_attacker = {
			limit = { NOT = { character = attacker } }
			hidden_effect = { participation_scaled_prestige = 100 }
		}
	}
	
	on_fail = {
		defender = {
			if = {
				limit = {
					claims_the_mandate_trigger = yes
				}
				mandate_boost_trivial_effect = yes
			}
		}
		
		attacker = {
			if = {
				limit = {
					claims_the_mandate_trigger = yes
				}
				mandate_penalty_trivial_effect = yes
			}
		}
	}

	on_fail_title = {
		attacker = {
			show_scope_change = no

			prestige = -100
		}

		defender = {
			show_scope_change = no

			participation_scaled_prestige = 50
		}

		any_defender = {
			limit = { NOT = { character = defender } }
			hidden_effect = { participation_scaled_prestige = 50 }
		}
	}

	on_reverse_demand = {
		attacker = {
			show_scope_change = no

			prestige = -200

			transfer_scaled_wealth = {
				to = defender
				value = 4.0
			}
		}

		defender = {
			show_scope_change = no

			participation_scaled_prestige = 100
		}

		any_defender = {
			limit = { NOT = { character = defender } }
			hidden_effect = { participation_scaled_prestige = 100 }
		}
		
		defender = {
			if = {
				limit = {
					claims_the_mandate_trigger = yes
				}
				mandate_boost_small_effect = yes
			}
		}
		
		attacker = {
			if = {
				limit = {
					claims_the_mandate_trigger = yes
				}
				mandate_penalty_small_effect = yes
			}
		}
	}

	attacker_ai_victory_worth = {
		factor = -1 # always accept
	}

	attacker_ai_defeat_worth = {
		factor = 100
	}

	defender_ai_victory_worth = {
		factor = -1 # always accept
	}

	defender_ai_defeat_worth = {
		factor = 100
	}

	ai_will_do = {
		factor = 0.9 # Prefer using CB's without on_add costs

		trigger = {
			trigger_if = {
				limit = {
					ROOT = {
						is_tribal = yes
					}

					FROM = {
						is_nomadic = yes
					}
				}

				any_direct_de_jure_vassal_title = {
					any_direct_de_jure_vassal_title = {
						OR = {
							holding_type = castle
							holding_type = city
							holding_type = tribal
						}
					}
				}
			}
			trigger_else_if = {
				limit = {
					ROOT = {
						is_nomadic = no
						is_tribal = no
					}

					FROM = {
						is_nomadic = yes
					}

				}

				any_direct_de_jure_vassal_title = {
					any_direct_de_jure_vassal_title = {
						OR = {
							holding_type = castle
							holding_type = city
						}
					}
				}
			}

			trigger_if = {
				limit = {
					ROOT = {
						has_character_flag = is_temujin
						has_landed_title = e_mongol_empire
						is_landed = yes
					}
				}

				FROM = {
					NOT = { has_landed_title = e_china_west_governor }
				}
			}
		}
	}
}

muslim_invasion = {

	can_use_gui = {
		# Attacker
		ROOT = {
			piety >= 1000

					OR = {
						gets_japanese_governments_trigger = yes
						AND = {
							gets_chinese_governments_trigger = yes
							is_tribal = no
							NOR = {
								government = successor_state_government

		ROOT = {

			NOR = {
				religion = FROM
				is_vassal_or_below_of = FROM

				has_character_modifier = muslim_invasion_cooldown
			}

			trigger_if = {
				limit = {
					ai = no
					multiplayer = yes
				}

				NOT = {
					has_game_rule = {
						name = multiplayer_invasions
						value = off
					}
				}
			}

			# These use 'tribal_invasion' instead
			#is_nomadic = no
			has_nomadic_tribal_government_trigger = no
			mercenary = no
		}
	}

	can_use_title = {
		# Muslims require the target to have at least 6 counties within the target kingdom
		FROM = {
			num_of_realm_counties = {
				value >= 6
				title = PREV
			}
		}

		# The attacker needs at least one county in the target kingdom, or a border
		any_direct_de_jure_vassal_title = {		# duchy level
			any_direct_de_jure_vassal_title = {	# county level
				OR = {
					holder_scope = { # Own a county
						OR = {
							character = ROOT
							is_vassal_or_below_of = ROOT
						}
					}
					
					AND = {
						holder_scope = {
							OR = {
								character = FROM
								is_vassal_or_below_of = FROM
							}
						}
						location = {
							any_neighbor_province = { # or share a border
								owner = {
									OR = {
										character = ROOT
										is_vassal_or_below_of = ROOT
									}
								}
							}
						}
					}
				}
			}
		}
	}

	is_valid = {
		# Attacker
		ROOT = {
			religion_group = muslim
			NOR = {
				religion = yazidi
				religion = druze
				religion = FROM
				AND = {
					any_liege = {
						gets_japanese_governments_trigger = yes
					}
					FROM = {
						OR = {
							is_the_tenno_trigger = yes
							is_ryukyuan_tenno_trigger = yes
						}
					}
				}
			}
		}
	}

	is_valid_title = {
		# Defender
		FROM = {
			num_of_realm_counties = {
				value >= 1
				title = PREV
			}
		}
	}

	on_add = {
		attacker = {
			show_scope_change = no

			piety = -1000
			
			if = {
				limit = {
					has_game_rule = {
						name = invasions
						value = limited
					}
				}
				
				add_character_modifier = {
					name = muslim_invasion_cooldown
					years = 50
					hidden = yes
				}
			}
		}

		fire_haruspicy_event_effect = yes
		anger_china_effect = yes
	}

	on_success = {
		attacker = {
			show_scope_change = no

			prestige = 200
			participation_scaled_prestige = 200

			if = {
				limit = {
					uses_decadence = yes

				}

				participation_scaled_decadence = -20
			}

			hidden_effect = {
				if = {

	truce_days = 3650

	allowed_to_target_tributaries = no

				limit = {
					FROM = {
						OR = {
							controls_religion = yes
							real_tier = emperor
						}
					}
				}
				piety >= 1000
			}
			trigger_else_if = {

					FROM = {
						real_tier = king
					}
				}
				piety >= 500
			}
			trigger_else = {
				piety >= 250
			}
			
			china_forbids_war_trigger = no
			
			NOT = {
				trait = decadent
			}
			
			NOT = {
				any_liege = {
