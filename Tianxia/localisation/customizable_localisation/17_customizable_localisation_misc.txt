# Returns GetBestName or "me" depending on whether FROM is the target character
defined_text = {
	name = GetAppropriateNameOrMeFrom
	use_first_valid = yes
	
	text = {
		localisation_key = GETTHISBESTNAME
		trigger = {
			FROM = {
				NOT = {
					character = THIS
				}
			}
		}
	}
	
	text = {
		localisation_key = MEME
		trigger = {
			FROM = {
				character = THIS
			}
		}
	}
}

# Returns "jungle" or "forest"
defined_text = {
	name = GetJungleOrForest
	use_first_valid = yes
	
	text = {
		localisation_key = JUNGLE
		trigger = {
			capital_scope = {
				OR = {
					region = world_india
					region = world_southeast_asia
				}
			}
		}
	}
	
	text = {
		localisation_key = FOREST
		trigger = {
			capital_scope = {
				NOR = {
					region = world_india
					region = world_southeast_asia
				}
			}
		}
	}
}

defined_text = {

	use_first_valid = yes

		localisation_key = killer_or_dw_sacrifice
		trigger = {
			OR = {
				killer = {
					character = FROM
				}
				AND = {
					killer = {
						same_society_as = FROM
					}
					FROM = {
						is_devil_worshiper_trigger = yes
					}
				}
			}
		}
	}
	
	fallback_text = {
