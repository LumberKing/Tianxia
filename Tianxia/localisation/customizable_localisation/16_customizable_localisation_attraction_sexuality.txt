defined_text = {
	name = GetHeterosexualDesc
	use_first_valid = yes

	text = {
		trigger = {
			OR = {
				trait = romance_interest_female
				trait = romance_interest_male
			}
		}
		localisation_key = heterosexual_desc_heteroromantic
	}

	text = {
		trigger = {
			NOR = {
				trait = romance_interest_female
				trait = romance_interest_male
			}
		}
		localisation_key = heterosexual_desc_aromantic
	}
	
	fallback_text = {
		localisation_key = heterosexual_desc_default
	}
}
