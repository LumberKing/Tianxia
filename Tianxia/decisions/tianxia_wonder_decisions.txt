	start_colosseum_restoration = {
		only_independent = yes
		is_high_prio = yes
		ai_check_interval = 3
		
		potential = {
			has_landed_title = e_roman_empire
			government = roman_imperial_government # Nobody else need apply
			has_global_flag = colosseum_possible
			completely_controls_c_roma_trigger = yes
			
			c_roma = {
				location = {
					NOT = {
						has_wonder = wonder_amphitheater
					}
				}
			}
			
			has_game_rule = {
				name = wonder_rule
				value = on
			}
		}
		
		allow = {
			has_regent = no
			
			has_secret_religion = no
			
			war = no
			
			wealth >= 1000
			
			prestige >= 500
			
			OR = {
				culture = roman
				culture = italian
				culture = greek
				culture = atlantean
			}
			
			religion_openly_hellenic_or_reformed_trigger = yes
			
			NOT = {
				has_religion_feature = religion_peaceful
			}
			
			trigger_if = {
				limit = {
					c_roma = {
						location = {
							has_wonder = yes
						}
					}
				}
				
				c_roma = {
					location = {
						wonder = {
							is_damaged = yes
						}
					}
				}
			}
		}
		
		effect = {
			clr_global_flag = colosseum_possible
			
			wealth = -1000
			
			c_roma = {
				location = {
					if = {
						limit = {
							has_wonder = yes
						}
						
						destroy_wonder = yes
					}
					
					add_wonder = wonder_colosseum
					
					custom_tooltip = {
						text = colosseum_can_be_repaired
						
						wonder = {
							# It was completed, once upon a time...
							add_wonder_stage = 1

							add_wonder_stage = 1
							
							# ... but needs some work...
							set_wonder_damaged = yes
							
							#add_custom_history = TEXT_RESTORED_TO_PROMINENCE
						}
					}
				}
			}
		}
		
		ai_will_do = {
			factor = 1
			
			trigger = {
				NOT = {
					trait = greedy
				}
			}
		}
	}
