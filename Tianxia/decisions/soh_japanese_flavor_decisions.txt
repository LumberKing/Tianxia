	celebrate_hanami = {
		only_playable = yes
		ai_check_interval = 12
		
		potential = {
			OR = {
				is_feudal = yes
				is_tribal = yes
				is_republic = yes
			}
			
			OR = { # TIANXIATODO: See if any of the below should get their own celebration
				has_japanese_religion_trigger = yes
				AND = {
					culture_group = japonic
					religion = buddhist
				}
			}
			
			NOT = {
				has_alternate_start_parameter = {
					key = religion_names
					value = random
				}
			}

			num_of_vassals >= 1
			prisoner = no
			
			NOT = {
				has_character_modifier = hanami_in_progress
			}
			
			OR = {
				NOT = {
					has_character_flag = recent_hanami
				}
				had_character_flag = { flag = recent_hanami days = 365 }
			}
			
			NOT = {
				liege = {
					has_character_modifier = hanami_in_progress
				}
			}
		}
		
		allow = {
			war = no
			prisoner = no
			is_inaccessible_or_incapable_trigger = no
			
			month >= 2 # During or after March
			month < 5 # Before June

			trigger_if = {
				limit = {
					NOT = {
						any_demesne_title = {
							lower_real_tier_than = DUKE
							is_occupied = no
						}
					}
				}

				custom_tooltip = {
					text = UNOCCUPIED_DEMESNE_TITLE
					always = no
				}
			}
			
			NOR = {
				trait = treasure_fleet_preparation
				trait = penglai_expedition_preparation
			}
		}
		
		effect = {
			wealth = -25
			
			add_character_modifier = {
				name = hanami_in_progress
				duration = -1
			}
			
			set_character_flag = recent_minor_decision
			set_character_flag = do_not_disturb
			
			hidden_tooltip = {
				character_event = { id = soh.3901 }
			}
			if = {
				limit = {
					has_dlc = "Reapers"
				}
				custom_tooltip = {
					text = capital_prospers_custom_tooltip
					hidden_tooltip = {
						capital_scope = {
							change_variable = { which = prosperity_value value = 8 }
						}
					}
				}
			}
		}
		
		ai_will_do = {
			factor = 1

			trigger = {
				wealth > 200
				diplomacy >= 10
				NOT = { trait = shy }

				trigger_if = {
					limit = { has_character_flag = recent_minor_decision }

					had_character_flag = {
						flag = recent_minor_decision
						years >= 2
					}
				}
			}
		}

			}
			
			OR = {
				independent = yes
				prestige >= 250
			}
		}
		
		effect = {
			if = {
				limit = {
					independent = no
				}
				prestige = -250
			}
			
			capital_scope = {
				PREV = {
					if = {
						limit = {
							ROOT = {
								higher_real_tier_than = duke
							}
						}
						
						spawn_unit = {
							owner = ROOT
							province = PREV
							home = PREV
							troops = {
								light_cavalry = { 400 400 }
								horse_archers = { 400 400 }
								heavy_infantry = { 200 200 }
								pikemen = { 1000 1000 }
								archers = { 1000 1000 }
							}
							
							earmark = warrior_bands_bloodline_forces
							maintenance_multiplier = 0.25
						}
						
						spawn_unit = {
							owner = ROOT
							province = PREV
							home = PREV
							troops = {
								light_cavalry = { 400 400 }
								horse_archers = { 400 400 }
								heavy_infantry = { 200 200 }
								pikemen = { 1000 1000 }
								archers = { 1000 1000 }
							}
							
							earmark = warrior_bands_bloodline_forces
							maintenance_multiplier = 0.25
						}
					}
					
					else_if = {
						limit = {
							ROOT = {
								tier = duke
							}
						}
						
						spawn_unit = {
							owner = ROOT
							province = PREV
							home = PREV
							troops = {
								light_cavalry = { 400 400 }
								horse_archers = { 400 400 }
								heavy_infantry = { 200 200 }
								pikemen = { 1000 1000 }
								archers = { 1000 1000 }
							}
							
							earmark = warrior_bands_bloodline_forces
							maintenance_multiplier = 0.25
						}
					}
					
					else = {
						spawn_unit = {
							owner = ROOT
							province = PREV
							home = PREV
							troops = {
								light_cavalry = { 200 200 }
								horse_archers = { 200 200 }
								heavy_infantry = { 100 100 }
								pikemen = { 500 500 }
								archers = { 500 500 }
							}
							
							earmark = warrior_bands_bloodline_forces
							maintenance_multiplier = 0.25
						}
					}
				}
			}
			
			hidden_tooltip = { 
				add_character_modifier = { 
					modifier = raised_warrior_band 
					duration = -1 #Removed on_action when player is back at peace
					hidden = yes
				} 
			}
		}
		
		ai_will_do = {
			factor = 1
		}
	}
