##### Fountain of Youth age fix

# on_yearly_pulse, age the character one year
character_event = {
	id = tianxia.1406
	hide_window = yes
	
	is_triggered_only = yes # on_yearly_pulse
	
	has_character_flag = fountain_of_youth_setup
	
	trigger = {
		is_actual_immortal_trigger = no
	}
	
	immediate = {
		fountain_of_youth_aging_effect = yes
		
		if = {
			limit = {
				OR = {
					practical_age >= 50
					is_sick_or_injured_trigger = yes
				}
			}
			
			random_list = {
				99 = {
					trigger = {
						OR = {
							AND = {
								practical_age < 65
								health >= 4
							}
							AND = {
								practical_age < 70
								health >= 4.5
							}
							AND = {
								practical_age < 75
								health >= 5
							}
							AND = {
								practical_age < 80
								health >= 5.5
							}
							AND = {
								practical_age < 85
								health >= 6
							}
							AND = {
								practical_age < 90
								health >= 6.5
							}
							AND = {
								practical_age < 95
								health >= 7
							}
							health >= 7.5
						}
					}
					
					mult_modifier = {
						factor = 0.85
						practical_age >= 70
					}
					
					mult_modifier = {
						factor = 0.85
						practical_age >= 75
					}
					
					mult_modifier = {
						factor = 0.85
						practical_age >= 80
					}
					
					mult_modifier = {
						factor = 0.85
						practical_age >= 85
					}
					
					mult_modifier = {
						factor = 0.85
						practical_age >= 90
					}
					
					mult_modifier = {
						factor = 0.85
						practical_age >= 95
					}
					
					mult_modifier = {
						factor = 0.85
						practical_age >= 100
					}
					
					mult_modifier = {
						factor = 0.9
						health < 6.5
					}
					
					mult_modifier = {
						factor = 0.9
						health < 6
					}
					
					mult_modifier = {
						factor = 0.9
						health < 5.5
					}
					
					mult_modifier = {
						factor = 0.9
						health < 5
					}
					
					mult_modifier = {
						factor = 0.9
						health < 4.5
					}
					
					# Survived
				}
				
				1 = {
					mult_modifier = {
						factor = 2
						is_sick_or_injured_trigger = yes
					}
					
					mult_modifier = {
						factor = 5
						has_epidemic_disease_trigger = yes
					}
					
					character_event = { id = tianxia.1407 days = 1 random = 364 } # Death by natural causes
				}
			}
		}
	}
}

# Delayed natural causes death for old characters that drank from the Fountain of Youth
character_event = {
	id = tianxia.1407
	hide_window = yes
	
	is_triggered_only = yes
	
	trigger = {
		is_actual_immortal_trigger = no
		OR = {
			practical_age >= 50
			is_sick_or_injured_trigger = yes
		}
	}
	
	immediate = {
		if = {
			limit = {
				has_deadly_trait_trigger = yes
			}
			
			death = {
				death_reason = death_trait
			}
		}
		
		else = {
			death = {
				death_reason = death_natural
			}
		}
	}
}

# on_yearly_pulse for the AI to consider backing a claimant if their liege is independent/a king AND they're a duke/king
character_event = {
	id = tianxia.1410
	hide_window = yes
	
	is_triggered_only = yes # on_yearly_pulse
	
	only_playable = yes
	min_age = 16
	prisoner = no
	only_capable = yes
	
	trigger = {
		ai = yes
		independent = no
		higher_real_tier_than = count
		liege = {
			OR = {
				independent = yes
				real_tier = king
			}
		}
		holy_order = no
		in_seclusion = no
		NOT = {
			trait = in_hiding
		}
		NOT = {
			in_faction = faction_claimant
		}
		
		liege = {
			is_vice_royalty = no

			OR = {
				is_feudal = yes
				is_tribal = yes
				is_nomadic = yes
			}
			
			trigger_if = {
				limit = {
					has_landed_title = e_mongol_empire
					culture = mongol
				}
				
				NAND = {
					OR = {
						AND = {
							religion = ROOT
							culture = ROOT
						}
						dynasty = ROOT
					}
					year < 1350
				}
			}
			
			trigger_else_if = {
				limit = {
					has_landed_title = e_aztec_empire
					culture = nahua
				}
				
				NAND = {
					religion = ROOT
					culture = ROOT
					year < 1350
				}
			}
			
			trigger_else_if = {
				limit = {
					has_landed_title = e_inca_empire
					culture = quechua
				}
				
				NAND = {
					religion = ROOT
					culture = ROOT
					year < 1350
				}
			}

			NOT = {
				any_demesne_title = {
					temporary = yes
				}
			}
			
			any_demesne_title = {
				OR = {
					holder_scope = {
						real_tier = ROOT
					}
					tier = king
				}
				NOT = {
					title = k_chrysanthemum_throne
				}
					
				trigger_if = {
					limit = {
						holder_scope = {
							government = chinese_imperial_government
						}
					}

						claimed_by = ROOT

							OR = {
								has_character_modifier = mandate_of_heaven_2
								has_character_modifier = mandate_of_heaven_1

						}

							PREV = {
								holder_scope = {
									dynasty = PREVPREV
								}
							}
						}
					}
				}
				
				OR = {
					AND = {
						holder_scope = {
							NOT = {
								de_jure_vassal_or_below = ROOT
							}
						}
						is_primary_holder_title = yes
					}
					AND = {
						is_primary_holder_title_tier = yes # Changed from is_primary_holder_title, since secondary titles might matter
						de_jure_vassal_or_below = ROOT
					}
					claimed_by = ROOT
				}

					limit = {

							OR = {
								religion = catholic

					}

							faction_exists = {
								faction = faction_antiking
								title = PREV
							}
						}
					}
				}

					limit = {

							has_law = succ_feudal_elective

							character = ROOT
						}
					}

					is_valid_claimant_prev_no_leader_trigger = yes

						is_ryukyuan_tenno_trigger = yes
					}

			}

			limit = { has_dlc = "Conclave" }

				is_voter = no
				is_nomadic = yes
				is_tribal = yes

					OR = {
