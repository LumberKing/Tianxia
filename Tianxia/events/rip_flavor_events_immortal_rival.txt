				}

					mult_modifier = {
						factor = 2
						is_wicked = yes
					}

						factor = 2

							trait = lunatic
							trait = cruel
							trait = wroth
							trait = cannibal
							trait = impaler
						}
					}

					set_character_flag = immortal_rival_goodguy
				}

				100 = {
					mult_modifier = {
						factor = 2
						combat_rating >= 30
					}

					mult_modifier = {
						factor = 1.5
						combat_rating >= 40
					}

					set_character_flag = immortal_rival_shieldmaiden
				}

				25 = {
					mult_modifier = {
						factor = 15
						any_courtier = {
							trait = horse
							is_actually_incapable_trigger = no
							prisoner = no
						}
					}

						factor = 10

						any_courtier = {
							trait = horse
							is_actually_incapable_trigger = no
