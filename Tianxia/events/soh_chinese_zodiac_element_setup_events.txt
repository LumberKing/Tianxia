# Events related to handouts of Chinese zodiac/element traits

namespace = soh

# IDs 13201-13300 reserved


##### Trait handouts 

# on_startup event
character_event = {
	id = soh.13201
	hide_window = yes
	
	is_triggered_only = yes # on_startup

	trigger = {
		NOT = {
			has_global_flag = chinese_zodiac_element_setup
		}
		owns = event_target:pulse_province
	}
	
	immediate = {
		set_global_flag = chinese_zodiac_element_setup
		
		any_character = {
			limit = {
				religion = taoist
				NOT = {
					has_character_flag = chinese_zodiac_set
				}
			}
			
			set_this_chinese_year_zodiac_element_effect = yes
		}
	}
}
