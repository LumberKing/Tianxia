		trigger = { has_artifact = i_ching }

	has_dlc = "Mystics"

	trigger = { has_a_taoist_divination_modifier_trigger = no }

		name = EVTOPTA_SOH_13301

		set_character_flag = battle_focused_divination_flag
	}

	option = { # I want to know where my relationships will lead
		name = EVTOPTB_SOH_13301

		set_character_flag = relationship_focused_divination_flag
	}

	option = { # I want to find out how to increase my power
		name = EVTOPTC_SOH_13301

		set_character_flag = power_hungry_divination_flag
	}

	after = {
		character_event = {
			id = soh.13302
			days = 5
			random = 1
		}
	}
}

# The outcome is divined
character_event = {
	id = soh.13302
	picture = GFX_evt_dark_prayer
	border = GFX_event_normal_frame_religion

	desc = {
		text = EVTDESCA_SOH_13302_BATTLE
		trigger = { has_character_flag = battle_focused_divination_flag }
	}
	desc = {
		text = EVTDESCB_SOH_13302_BATTLE
		trigger = { has_character_flag = battle_focused_divination_flag }
	}
	desc = {
		text = EVTDESCC_SOH_13302_BATTLE
		trigger = { has_character_flag = battle_focused_divination_flag }
	}

	desc = {
		text = EVTDESCA_SOH_13302_RELATIONSHIP
		trigger = { has_character_flag = relationship_focused_divination_flag }
	}
	desc = {
		text = EVTDESCB_SOH_13302_RELATIONSHIP
		trigger = { has_character_flag = relationship_focused_divination_flag }
	}
	desc = {
		text = EVTDESCC_SOH_13302_RELATIONSHIP
		trigger = { has_character_flag = relationship_focused_divination_flag }
	}

	desc = {
		text = EVTDESCA_SOH_13302_POWER
		trigger = { has_character_flag = power_hungry_divination_flag }
	}
	desc = {
		text = EVTDESCB_SOH_13302_POWER
		trigger = { has_character_flag = power_hungry_divination_flag }
	}
	desc = {
		text = EVTDESCC_SOH_13302_POWER
		trigger = { has_character_flag = power_hungry_divination_flag }
	}

	is_triggered_only = yes

	has_dlc = "Mystics"
	
	religion = taoist

	trigger = {
		has_a_taoist_divination_modifier_trigger = no
	}

	immediate = {
		random_list = {
			50 = {
				mult_modifier = {
					factor = 1.5
					has_artifact = i_ching
				}
				
				set_character_flag = positive_reaction_to_divination_flag
			}

			50 = {
				set_character_flag = negative_reaction_to_divination_flag
			}
		}
	}

	option = {
		name = EVTOPTA_SOH_13302

		trigger = { has_character_flag = positive_reaction_to_divination_flag }

		trigger_switch = {
			on_trigger = has_character_flag

			battle_focused_divination_flag = {
				add_character_modifier = {
					name = good_battle_taoist_divination
					duration = -1
				}
			}
			relationship_focused_divination_flag = {
				add_character_modifier = {
					name = good_relationship_taoist_divination
					duration = -1
				}
			}
			power_hungry_divination_flag = {
				add_character_modifier = {
					name = good_power_taoist_divination
					duration = -1
				}
			}
		}
	}

	option = {
		name = EVTOPTB_SOH_13302

		trigger = { has_character_flag = negative_reaction_to_divination_flag }

		trigger_switch = {
			on_trigger = has_character_flag

			battle_focused_divination_flag = {
				add_character_modifier = {
					name = bad_battle_taoist_divination
					duration = -1
				}
			}
			relationship_focused_divination_flag = {
				add_character_modifier = {
					name = bad_relationship_taoist_divination
					duration = -1
				}
			}
			power_hungry_divination_flag = {
				add_character_modifier = {
					name = bad_power_taoist_divination
					duration = -1
				}
			}
		}
	}

	after = {
		hidden_effect = {
			random_list = { # For sending fallbacks, in case nothing else triggers from the divination modifiers during the next few years
				33 = {
					character_event = {
						id = soh.13324
						days = 1500
						random = 300
					}
				}

				33 = {
					character_event = {
						id = soh.13325
						days = 1500
						random = 300
					}
				}

				33 = {
					character_event = {
						id = soh.13326
						days = 1500
						random = 300
					}
				}
			}

			remove_character_modifier = taoist_divination_cooldown
			clr_character_flag = battle_focused_divination_flag
			clr_character_flag = relationship_focused_divination_flag
			clr_character_flag = power_hungry_divination_flag
		}
	}
}

# Gain a leadership trait
character_event = {
	id = soh.13303
	title = the_battle_at_location
	picture = GFX_evt_battle
	border = GFX_event_normal_frame_war

	desc = {
		text = EVTDESCA_SOH_13303
		trigger = { has_character_modifier = good_battle_taoist_divination }
	}
	desc = {
		text = EVTDESCB_SOH_13303
		trigger = { has_character_modifier = bad_battle_taoist_divination }
	}

	is_triggered_only = yes # on_combat_pulse

	has_dlc = "Mystics"

	trigger = {
		has_battle_taoist_divination_modifier_trigger = yes
		can_have_more_leadership_traits = yes

		NAND = {
			trait = holy_warrior
			trait = unyielding_leader
			trait = inspiring_leader
		}
	}

	immediate = {
		disallow_new_battle_events_effect = yes

		random_list = {
			33 = {
				trigger = { NOT = { trait = holy_warrior } }
				set_character_flag = gain_holy_warrior
			}

			33 = {
				trigger = { NOT = { trait = unyielding_leader } }
				set_character_flag = gain_unyielding_leader
			}

			33 = {
				trigger = { NOT = { trait = inspiring_leader } }
				set_character_flag = gain_inspiring_leader
			}
		}
	}

	option = { # Gain Holy Warrior
		name = EVTOPTA_SOH_13303

		trigger = { has_character_flag = gain_holy_warrior }

		add_trait = holy_warrior
	}

	option = { # Gain Unyielding
		name = EVTOPTB_SOH_13303

		trigger = { has_character_flag = gain_unyielding_leader }

		add_trait = unyielding_leader
	}

	option = { # Gain Inspiring Leader
		name = EVTOPTC_SOH_13303

		trigger = { has_character_flag = gain_inspiring_leader }

		add_trait = inspiring_leader
	}

	after = {
		if = {
			limit = { has_character_modifier = good_battle_taoist_divination }
			upgrade_diviner_modifier_effect = yes
		}
		else = {
			downgrade_diviner_modifier_effect = yes
		}

		remove_taoist_divination_modifiers_effect = yes
		clr_character_flag = gain_holy_warrior
		clr_character_flag = gain_unyielding_leader
		clr_character_flag = gain_inspiring_leader
		allow_new_battle_events_effect = yes
	}
}

	has_dlc = "Mystics"

	trigger = {

	}

	immediate = {

	}

	option = { # At least I was right
		name = EVTOPTA_SOH_13304

		trigger = { has_character_modifier = bad_battle_taoist_divination }

		upgrade_diviner_modifier_effect = yes
		
		add_trait_wounded_effect = yes
	}

	option = { # Curses! Where's the glory in this?!
		name = EVTOPTB_SOH_13304

		trigger = { has_character_modifier = good_battle_taoist_divination }

		downgrade_diviner_modifier_effect = yes
		
		add_trait_wounded_effect = yes
	}

	after = {
		random_list = {
			40 = { add_trait_wounded_effect = yes }

			40 = {
				trigger = {
					has_dlc = "Reapers"
					NOT = { trait = one_legged }
				}

				add_trait_one_legged_effect = yes
			}

			5 = {
				death = { death_reason = death_battle }
			}
		}

		remove_taoist_divination_modifiers_effect = yes
		allow_new_battle_events_effect = yes
	}
}

# Newborn sends event to parents
character_event = {
	id = soh.13307

	is_triggered_only = yes
	hide_window = yes

	has_dlc = "Mystics"

	trigger = {
		OR = {
			father = { has_relationship_taoist_divination_modifier_trigger = yes }
			mother = { has_relationship_taoist_divination_modifier_trigger = yes }
		}

		OR = {
			is_smart_trigger = yes
			is_strong_trigger = yes
			has_negative_congenital_trigger = yes
		}
	}

	weight_multiplier = {
		factor = 1

		mult_modifier = {
			factor = 5

			OR = {
				has_character_modifier = bad_relationship_taoist_divination
				has_character_modifier = good_relationship_taoist_divination
			}
		}
	}

	immediate = {
		if = {
			limit = { has_negative_congenital_trigger = no }

			mother = { character_event = { id = soh.13305 days = 1 } }
			father = { character_event = { id = soh.13305 days = 1 } }
		}
		else = {
			mother = { character_event = { id = soh.13306 days = 1 } }
			father = { character_event = { id = soh.13306 days = 1 } }
		}
	}
}

# PING/Set-up event: Someone dies - is this what you foretold?
character_event = {
	id = soh.13308

	hide_window = yes
	is_triggered_only = yes # on_death, random_events

	trigger = {
		death_no_trace_trigger = no
		#death_in_china_trigger = no

		OR = {
			any_lover = {
				is_actually_incapable_trigger = no
				has_relationship_taoist_divination_modifier_trigger = yes
			}

			any_friend = {
				is_actually_incapable_trigger = no
				has_relationship_taoist_divination_modifier_trigger = yes
			}

			trigger_if = {
				limit = { is_adult = yes }

				any_rival = {
					is_actually_incapable_trigger = no
					has_relationship_taoist_divination_modifier_trigger = yes
				}
			}

			any_child = {
				is_adult = yes
				is_actually_incapable_trigger = no
				has_relationship_taoist_divination_modifier_trigger = yes
			}

			mother = {
				is_actually_incapable_trigger = no
				has_relationship_taoist_divination_modifier_trigger = yes
			}

			father = {
				is_actually_incapable_trigger = no
				has_relationship_taoist_divination_modifier_trigger = yes
			}
		}
	}

	option = {
		save_event_target_as = soh_divination_dead

		# Lover gets event
		random_lover = {
			limit = {
				is_actually_incapable_trigger = no
				has_relationship_taoist_divination_modifier_trigger = yes
			}

			character_event = { id = soh.13309 days = 1 }
		}

		# Friend gets event
		random_friend = {
			limit = {
				is_actually_incapable_trigger = no
				has_relationship_taoist_divination_modifier_trigger = yes
			}

			character_event = { id = soh.13310 days = 1 }
		}

		# Rival gets event
		random_rival = {
			limit = {
				is_actually_incapable_trigger = no
				has_relationship_taoist_divination_modifier_trigger = yes
			}

			character_event = { id = soh.13311 days = 1 }
		}

		# Adult child gets event
		random_child = {
			limit = {
				is_adult = yes
				is_actually_incapable_trigger = no
				has_relationship_taoist_divination_modifier_trigger = yes
			}

			character_event = { id = soh.13312 days = 1 }
		}

		# Parent
		mother = {
			if = {
				limit = {
					is_actually_incapable_trigger = no
					has_relationship_taoist_divination_modifier_trigger = yes
				}

				character_event = { id = soh.13313 days = 1 }
			}
		}

		father = {
			if = {
				limit = {
					is_actually_incapable_trigger = no
					has_relationship_taoist_divination_modifier_trigger = yes
				}

				character_event = { id = soh.13313 days = 1 }
			}
		}
	}
}

# 2: gating event, to send event or clearing flag for the new holder
# ROOT is new holder of title
# FROMFROM is old holder of title
character_event = {
	id = soh.13318

	hide_window = yes
	is_triggered_only = yes # on_new_holder(_inheritance/usurpation), 1 day delay to allow soh.13317 to fire first.

	trigger = {
		FROMFROM = { is_alive = yes }
	}

	immediate = {
		if = {
			limit = { has_character_flag = was_granted_title_voluntarily }
			clr_character_flag = was_granted_title_voluntarily
		}
		else = {
			FROMFROM = { character_event = { id = soh.13319 days = 1 } }
		}
	}
}
