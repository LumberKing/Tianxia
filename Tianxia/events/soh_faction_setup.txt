# Duan Siping
character_event = {
	id = soh.13401
	hide_window = yes
	
	is_triggered_only = yes # on_startup
	
	only_playable = yes
	
	has_character_flag = is_duan_siping
	
	trigger = {
		start_date == 936.8.7 # IC
		NOT = {
			has_global_flag = 936_dali_factions
		}
	}
	
	immediate = {
		set_global_flag = 936_dali_factions
		remove_title = job_chancellor
		remove_title = job_marshal
		remove_title = job_treasurer
		remove_title = job_spymaster
		remove_title = job_spiritual
		remove_title = title_councilmember_king
		start_faction = {
			faction = faction_claimant
			title = k_dali
			thirdparty = ROOT
		}
		start_faction = {
			faction = faction_powerful_vassal_takeover
			title = k_dali
		}
	}
}
