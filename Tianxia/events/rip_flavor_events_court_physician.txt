		is_nomadic = no

			limit = { health >= 5 }

				trait = stressed

		}

			trait = slothful
			has_character_flag = stress_relief

			has_minor_title = title_court_physician

			OR = {
				has_character_flag = is_court_physician
				trait = physician
			}
		}
	}

	weight_multiplier = {
		factor = 1

		mult_modifier = {
			factor = 2
			trait = diligent
		}

		mult_modifier = {
			factor = 2
			trait = stressed
		}

		mult_modifier = {
			factor = 1.5

			any_courtier_or_vassal = {
				has_minor_title = title_court_physician
				learning >= 16
			}
		}

		mult_modifier = {
			factor = 0.2

			any_courtier_or_vassal = {
				has_minor_title = title_court_physician
				learning < 14
			}
		}

		mult_modifier = {
			factor = 0.1

			any_courtier_or_vassal = {
				has_minor_title = title_court_physician
				learning < 10
			}
		}
	}

	immediate = {
		random_courtier_or_vassal = {
			limit = { has_minor_title = title_court_physician }
			save_event_target_as = physician_target
		}

		set_character_flag = stress_relief

		name = EVTOPTA_RIP_4010

		prestige = -50

			chance = 50
			remove_trait_stressed_effect = yes
		}

		remove_trait_silently_diligent_effect = yes

			name = taking_it_easy

		}

	option = {
		name = EVTOPTB_RIP_4010

		}

		any_courtier_or_vassal = {
			is_tribal = no
			is_nomadic = no
			has_minor_title = title_court_physician
			NOT = { has_character_flag = spent_time_at_hospital }
			ai = yes
		}
	}

	immediate = {
		random_demesne_province = {
			limit = {
				owned_by = ROOT
				hospital_level >= 1

					hospital_has_building = library_1

					hospital_has_building = medical_academy_1
				}
			}
			save_event_target_as = hospital_location
		}

		random_courtier_or_vassal = {
			limit = { has_minor_title = title_court_physician }
			save_event_target_as = learning_physician
		}
	}

	option = { # Accept
		name = EVTOPTA_RIP_4530

		scaled_wealth = -0.05

		hidden_effect = {
			event_target:learning_physician = {
				set_character_flag = spent_time_at_hospital

					days = 30
					random = 30

		}
	}

	option = { # Refuse
		name = EVTOPTB_RIP_4530
	}
