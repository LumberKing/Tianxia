	hide_window = yes
	
	is_triggered_only = yes
	
	immediate = {
		if = {
			limit = {
				holds_favor_on = event_target:target_vassal
			}
			
			# Call in the Favor; the player doesn't get to pick the other option, so the AI also doesn't get it
			
			#log = "[This.GetBestName] meddled in [From.GetBestName]'s war against [target_defender.GetBestName] using a Favor!"
			
			event_target:target_vassal = {
				reverse_remove_favor = ROOT

			}

			random_list = {
