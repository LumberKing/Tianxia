# Used for AI factioning through events (workaround for "elective" titles where not all claimants can be candidates)

#claimant_score

claimant_score = {
	mult_modifier = {
		factor = 0.1

		# If we are subjugated, tend to unite behind one single claimant
		event_target:claimed_title = {
			is_conquered = yes
			NOT = { conquest_culture = event_target:potential_faction_leader }
		}

		event_target:faction_target = {
			faction_exists = {
				faction = faction_claimant
				title = event_target:claimed_title
			}
		}

		culture = event_target:potential_faction_leader

		event_target:potential_faction_leader = {
			opinion_diff = {
				first = PREV
				second = liege
				value >= 10 # I like the Claimant more than the current ruler
				as_if_liege = yes
			}
		}
	}

	mult_modifier = {
		factor = 0.5

		event_target:faction_target = {
			any_owned_bloodline = {

			}
		}
	}

	mult_modifier = {
		factor = 0.2

	}

	# Or if vassal is under the yoke of the wrong kind of liege.
	mult_modifier = {
		factor = 3

		event_target:potential_faction_leader = {
			OR = {
				has_religion_feature = religion_patriarchal
				has_religion_feature = religion_feature_batak
				has_religion_feature = religion_feature_yi
				AND = {
					religion = confucian
					gender_equality_trigger = no
					religions_are_standard_trigger = yes
				}
			}
			is_female = yes
		}

		is_female = no
	}

	mult_modifier = {

	}

					}
				}

				has_law = status_of_women_4

				has_game_rule = {
					name = gender
					value = all
				}
			}
		}
	}
