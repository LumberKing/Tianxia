true_religion_has_incest = { # TIANXIATODO: Fix
	OR = {
		AND = {
			religions_are_standard_trigger = yes
			OR = {
				true_religion = zoroastrian
				true_religion = messalian
			}
		}
		true_religion_scope = {
			OR = {
				has_religion_feature = religion_feature_zun
				has_religion_feature = religion_holy_family
			}
		}
		true_religion_scope = {
			has_flag = incest_religion
		}
		# Technically also inca_pagan, inca_pagan_reformed
	}
}



could_have_chinese_government_trigger = {
	trigger_if = {
		limit = {
			has_chinese_eastern_religion_trigger = yes
		}
		
		has_chinese_eastern_religion_trigger = yes
	}
	
	trigger_else_if = {
		limit = {
			religion = buddhist
		}
		
		religion = buddhist
		trigger_if = {
			limit = {
				real_tier = emperor
			}
			
			tier = emperor
			has_chinese_or_historical_culture_for_chinese_imperial_trigger = yes
		}
		
		trigger_else = {
			has_chinese_or_historical_culture_trigger = yes
		}
	}
	
	trigger_else = {
		can_have_confucian_bureaucracy_and_eastern_imperial_only_religion_trigger = yes
	}
}

could_have_japanese_government_trigger = {
	trigger_if = {
		limit = {
			has_japanese_religion_trigger = yes
		}
		
		has_japanese_religion_trigger = yes
	}
	
	trigger_else_if = {
		limit = {
			religion = buddhist
		}
		
		religion = buddhist
		culture_group = japonic
	}
	
	trigger_else_if = {
		limit = {
			religion_openly_ryukyuan_or_reformed_trigger = yes
		}
		
		trigger_if = {
			limit = {
				religion = ryukyuan_pagan
			}
			
			religion = ryukyuan_pagan
		}
		
		trigger_else = {
			has_dlc = "Holy Fury"
			
			trigger_if = {
				limit = {
					controls_religion = yes
				}
				
				religion_has_divine_imperial_reformation_trigger = yes
			}
			
			trigger_else = {
				OR = {
					religion_has_japanese_government_reformation_trigger = yes
					religion_has_divine_imperial_reformation_trigger = yes
				}
			}
		}
	}
	
	trigger_else = {
		always = no
	}
}

has_proselytizing_religion_character_trigger = { # TIANXIATODO: Hook into assorted things
	OR = {
		AND = {
			religions_are_standard_trigger = yes
			OR = {
				religion_group = christian # All in favour
				# Other Eastern religions do not proselytize
				religion = buddhist # Happened historically from time to time
				# The Jewish religions frown on proselytizing
				AND = {
					religion_group = muslim
					NOR = { # Not really Muslim...
						religion = druze
						religion = yazidi
					}
				}
				# Most Mazdans don't care for proselytizing
				religion = manichaean
			}
		}
		
		
		has_religion_feature = religion_proselytizing
	}
}

has_proselytizing_religion_province_trigger = { # TIANXIATODO: Hook into assorted things
	OR = {
		AND = {
			religions_are_standard_trigger = yes
			OR = {
				religion_group = christian # All in favour
				# Other Eastern religions do not proselytize
				religion = buddhist # Happened historically from time to time
				# The Jewish religions frown on proselytizing
				AND = {
					religion_group = muslim
					NOR = { # Not really Muslim...
						religion = druze
						religion = yazidi
					}
				}
				# Most Mazdans don't care for proselytizing
				religion = manichaean
			}
		}
		
		county = {
			holder_scope = {
				any_playable_ruler = { # Not ideal, but better than nothing...
					religion = PREV
					has_religion_feature = religion_proselytizing
				}
			}
		}
	}
}
