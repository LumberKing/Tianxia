		text = completely_controls_donation_of_pepin
		completely_controls = d_latium
		completely_controls = d_spoleto
		completely_controls = d_ancona
		completely_controls = d_ferrara
		completely_controls = d_benevento
		NOT = {
			any_independent_ruler = {
				is_nomadic = yes
				NOT = {
					character = PREV # If we always know the character isn't a nomad, this can probably be removed
				}
				OR = {
					any_demesne_title = { # Demesne counties
						tier = count
						OR = {
							de_jure_liege_or_above = d_latium
							de_jure_liege_or_above = d_spoleto
							de_jure_liege_or_above = d_ancona
							de_jure_liege_or_above = d_ferrara
							de_jure_liege_or_above = d_benevento
						}
					}
					any_vassal = {
						is_nomadic = yes # Non-nomadic vassals and sub-vassals have holdings in their provinces, and would thus cause the completely_controls(_region) = <title/region> to fail
						any_demesne_title = { # Demesne counties
							tier = count
							OR = {
								de_jure_liege_or_above = d_latium
								de_jure_liege_or_above = d_spoleto

						}
					}
				}
			}
		}
	}

		text = completely_controls_roman_mauretania_caesariensis
		completely_controls_region = roman_mauretania_caesariensis
		NOT = {
			any_independent_ruler = {
				is_nomadic = yes
				NOT = {
					character = PREV # If we always know the character isn't a nomad, this can probably be removed
				}
				OR = {
					any_demesne_title = { # Demesne counties
						tier = count
						region = roman_mauretania_caesariensis
					}
					any_vassal = {
						is_nomadic = yes # Non-nomadic vassals and sub-vassals have holdings in their provinces, and would thus cause the completely_controls(_region) = <title/region> to fail
						any_demesne_title = { # Demesne counties
							tier = count
							region = roman_mauretania_caesariensis
						}
					}
				}
			}
		}
	}
}

		text = completely_controls_roman_khazaria
		completely_controls_region = roman_khazaria
		NOT = {
			any_independent_ruler = {
				is_nomadic = yes
				NOT = {
					character = PREV # If we always know the character isn't a nomad, this can probably be removed
				}
				OR = {
					any_demesne_title = { # Demesne counties
						tier = count
						region = roman_khazaria
					}
					any_vassal = {
						is_nomadic = yes # Non-nomadic vassals and sub-vassals have holdings in their provinces, and would thus cause the completely_controls(_region) = <title/region> to fail
						any_demesne_title = { # Demesne counties
							tier = count
							region = roman_khazaria
						}
					}
				}
			}
		}
	}
}
