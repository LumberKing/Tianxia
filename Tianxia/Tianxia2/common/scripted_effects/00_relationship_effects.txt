	# First remove any rivalry
	if = {
		limit = {
			is_rival = event_target:changing_relationship_target
		}
		remove_rival = event_target:changing_relationship_target
	}
	# Then remove any rivalry modifiers
	ex_rival_opinions_cleanup_effect = yes
	# Then remove any modifiers that should not be present on the lovers

	# And finally become lovers

start_romance_effect = {
	# First remove any rivalry
	if = {
		limit = {
			is_rival = event_target:changing_relationship_target
		}
		remove_rival_single_character_effect = yes
	}
	
	# Then remove any leftover bad stuff
	new_romance_opinions_cleanup_effect = yes
	
	# Then clean up lesser modifiers
	if = {
		limit = {
			has_opinion_modifier = {
				who = event_target:changing_relationship_target
				name = opinion_crush
			}
		}
		remove_opinion = {
			who = event_target:changing_relationship_target
			name = opinion_crush
		}
	}
	
	if = {
		limit = {
			has_opinion_modifier = {
				who = event_target:changing_relationship_target
				name = opinion_infatuated
			}
		}
		remove_opinion = {
			who = event_target:changing_relationship_target
			name = opinion_infatuated
		}
	}
	
	# Then add the correct romance type
	if = {
		limit = {
			is_adult = no
		}
		opinion = {
			who = event_target:changing_relationship_target
			name = opinion_childhood_sweetheart
		}
	}
	
	else = {
		if = {
			limit = {
				event_target:changing_relationship_target = {
					has_opinion_modifier = {
						who = PREV
						name = opinion_budding_romance
					}
				}
			}
			
			opinion = {
				who = event_target:changing_relationship_target
				name = opinion_mutual_budding_romance
			}
			
			event_target:changing_relationship_target = {
				remove_opinion = {
					who = PREV
					name = opinion_budding_romance
				}
				opinion = {
					who = PREV
					name = opinion_mutual_budding_romance
				}
			}
		}
		
		else = {
			opinion = {
				who = event_target:changing_relationship_target
				name = opinion_budding_romance
			}
		}
	}
}
