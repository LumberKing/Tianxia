set_up_sexuality_romance_effect = {
	# Only used for adult characters
	if = {
		limit = {
			NOT = {
				has_character_flag = sexuality_romance_set
			}
		}
		
		set_character_flag = sexuality_romance_set
		
		if = { # Unlikely to be the case, but we might as well extend the effect...
			limit = {
				OR = {
					has_any_romance_interest_trait_trigger = yes
					has_character_flag = aro_ace
				}
				has_any_sex_interest_trait_trigger = no
				has_any_sexuality_trait_trigger = no
			}
			
			add_trait = asexual
		}
		
		else_if = { # For characters we don't want to lock in a specific sexuality on but have pre-defined some sexual interest for
			limit = {
				has_any_sex_interest_trait_trigger = yes
				has_any_sexuality_trait_trigger = no
			}
			
			randomize_valid_sexuality_effect = yes
			randomize_valid_romance_effect = yes
		}
		
		else_if = { # Usual case for anyone we pre-define a specific sexuality for
			limit = {
				has_any_sexuality_trait_trigger = yes
			}
			
			hidden_tooltip = {
				trigger_switch = {
					on_trigger = trait
					
					heterosexual = {
						if = {
							limit = {
								is_female = yes
							}
							add_trait = sex_interest_male
							add_trait = sex_attraction_male
						}
						else = {
							add_trait = sex_interest_female
							add_trait = sex_attraction_female
						}
					}
					
					homosexual = {
						if = {
							limit = {
								is_female = yes
							}
							add_trait = sex_interest_female
							add_trait = sex_attraction_female
						}
						else = {
							add_trait = sex_interest_male
							add_trait = sex_attraction_male
						}
					}
					
					bisexual = {
						add_trait = sex_interest_male
						add_trait = sex_interest_female
						add_trait = sex_attraction_female
						add_trait = sex_attraction_male
					}
					
					asexual = {
						asexual_sex_interest_randomization_effect = yes
					}
				}
			}
			randomize_valid_romance_effect = yes
		}
		
		else = { # Nothing pre-defined
			randomize_sexuality_romance_effect = yes
		}
		
		attraction_cutoff_roll_effect = yes
	}
	
	add_all_relevant_attraction_giving_traits_effect = yes
	add_all_relevant_attraction_giving_modifiers_effect = yes
	
	update_fertility_penalty_effect = yes
}

				limit = {
					is_female = yes

				set_character_flag = sex_attraction_female_known
				set_character_flag = no_sex_attraction_male_known
				set_character_flag = no_romance_interest_male_known
			}
			else = {
				set_character_flag = sex_attraction_male_known
				set_character_flag = no_sex_attraction_female_known
				set_character_flag = no_romance_interest_female_known
			}
		}
		
		bisexual = {
			set_character_flag = sex_attraction_female_known
			set_character_flag = sex_attraction_male_known
		}
		
		asexual = {
			set_character_flag = no_sex_attraction_female_known
			set_character_flag = no_sex_attraction_male_known
		}
	}
	
	# Clean up sex attraction traits
	remove_trait = sex_attraction_female
	remove_trait = sex_attraction_male
	
	# Romance attraction trait cleanup
	if = {
		limit = {
			trait = romance_attraction_female
		}
		remove_trait = romance_attraction_female
		if = {
			limit = {
				NOT = {
					trait = romance_interest_female
				}
			}
			set_character_flag = romance_interest_female_unknown
		}
	}
	else_if = {
		limit = {
			NOR = {
				trait = romance_interest_female
				has_character_flag = romance_interest_female_unknown
				has_character_flag = no_romance_interest_female_known
			}
		}
		set_character_flag = no_romance_interest_female_unknown
	}
	
	if = {
		limit = {
			trait = romance_attraction_male
		}
		remove_trait = romance_attraction_male
		if = {
			limit = {
				NOT = {
					trait = romance_interest_male
				}
			}
			set_character_flag = romance_interest_male_unknown
		}
	}

		limit = {
			NOR = {
				trait = romance_interest_male
				has_character_flag = romance_interest_male_unknown
				has_character_flag = no_romance_interest_male_known
			}
		}
		set_character_flag = no_romance_interest_male_unknown
	}
	
	# Add "placeholder" trait if needed
	if = {
		limit = {
			has_any_sexuality_trait_trigger = no
		}
		add_trait = figuring_things_out
	}
	
	# See if something is figured out
	figuring_things_out_chance_effect = yes
}
